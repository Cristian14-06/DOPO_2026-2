3   import java.util.EnumSet;
4   import java.util.Set;

6   /**
7    * Machine controller implementing the Defensiva (defensive) profile prioritizing goal protection.
8    *
9    * @author Juan Espitia
10   * @author Cristian Salamanca
11   */
12  public class DefensiveMachine extends MachineController {
13      private static final long serialVersionUID = 1L;
14      private static final double HOME_GOAL_OFFSET = 70.0;
15      private static final double DEFEND_ZONE_RADIUS = 170.0;
16      private static final double BEHIND_BALL_OFFSET = 12.0;
17      private static final double MOVE_TOLERANCE = 10.0;
18      private static final double JUMP_MAX_DX = 30.0;
19      private static final double JUMP_HEAD_MARGIN_Y = 10.0;
20      private static final double KICK_MAX_DX = 28.0;
21      private static final double KICK_MAX_HEIGHT = 70.0;
22  
23      /**
24       * Constructs a defensive machine controller for the given pitch side.
25       *
26       * @param side Pitch side controlled by this AI
27       */
28      public DefensiveMachine(Side side) {
29          super(side);
30      }
31  
32      /**
33       * {@inheritDoc}
34       */
35      @Override
36      protected Set<PlayerAction> chooseActions(MatchView match, PlayerView me, BallView ball, PlayerView rival) {
37          Set<PlayerAction> actions = EnumSet.noneOf(PlayerAction.class);
38          double myX = me.getHeadShape().getX();
39          double goalX = ownGoalX(match);
40          double homeX = goalX + side.dir() * HOME_GOAL_OFFSET;
41          if(Math.abs(ball.getX() - goalX) > DEFEND_ZONE_RADIUS){
42              moveToward(myX, homeX, MOVE_TOLERANCE, actions);
43          }else{
44              double defendX = ball.getX() - side.dir()*BEHIND_BALL_OFFSET;
45              moveToward(myX, defendX, 4.0, actions);
46              if(shouldJumpForBall(me, ball, JUMP_MAX_DX, JUMP_HEAD_MARGIN_Y)) actions.add(PlayerAction.JUMP);
47              if(shouldKickBall(match, me, ball, KICK_MAX_DX, KICK_MAX_HEIGHT)
48                      && (ball.getX() - myX) * side.dir() > -10){
49                  actions.add(PlayerAction.KICK);
50              }
51          }
52          return actions;
53          
54          
55      }
56  }
