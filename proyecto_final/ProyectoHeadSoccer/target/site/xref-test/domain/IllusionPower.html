3   import java.util.ArrayList;
4   import java.util.List;

6   /**
7    * United States signature power shot: Illusion.
8    * Splits the visual ball into eight identical balls (seven decoys) for 2.5 seconds,
9    * inflicting the Confused status effect and dealing 30 base power damage.
10   *
11   * @author Juan Espitia
12   * @author Cristian Salamanca
13   */
14  public class IllusionPower extends Power {
15      private List<Circle> fakeBalls;
16      private int illusionTicksRemaining;
17  
18      /**
19       * Constructs USA's Illusion power.
20       */
21      public IllusionPower() {
22          super("Ilusión", 2.5);
23          this.fakeBalls = new ArrayList<>();
24          this.illusionTicksRemaining = 0;
25      }
26  
27      /**
28       * Launches the ball and spawns 7 illusionary decoy balls around it.
29       *
30       * @param ball Ball in play
31       * @param side Side of the shooting player
32       */
33      @Override
34      protected void shapeShot(Ball ball, Side side) {
35          if (ball == null) return;
36          double dir = (side == Side.LEFT) ? 1.0 : -1.0;
37          ball.setHorizontalSpeed(18.0 * dir);
38          ball.setVerticalSpeed(-5.0);
39  
40          fakeBalls.clear();
41          illusionTicksRemaining = (int) (getEffectDurationSeconds() * Physics.FPS);
42  
43          double bx = ball.getCircle().getX();
44          double by = ball.getCircle().getY();
45          double r = ball.getCircle().getRadius();
46  
47          for (int i = 0; i < 7; i++) {
48              double angle = i * (2 * Math.PI / 7);
49              double fx = bx + Math.cos(angle) * (18 + i * 4);
50              double fy = by + Math.sin(angle) * (15 + i * 3);
51              fakeBalls.add(new Circle(fx, fy, r));
52          }
53      }
54  
55      /**
56       * Updates illusion clones movement following the real ball with slight offsets.
57       * 
58       * @param realBall Main game ball
59       */
60      public void updateClones(Ball realBall) {
61          if (illusionTicksRemaining > 0 && realBall != null) {
62              illusionTicksRemaining--;
63              double bx = realBall.getCircle().getX();
64              double by = realBall.getCircle().getY();
65              for (int i = 0; i < fakeBalls.size(); i++) {
66                  Circle c = fakeBalls.get(i);
67                  double offsetAngle = (i * 0.9) + (illusionTicksRemaining * 0.1);
68                  c.setX(bx + Math.cos(offsetAngle) * (20 + i % 3 * 8));
69                  c.setY(by + Math.sin(offsetAngle) * (14 + i % 2 * 6));
70              }
71          } else {
72              fakeBalls.clear();
73          }
74      }
75  
76      /**
77       * Returns the list of active illusionary decoy circles.
78       *
79       * @return List of fake ball circles
80       */
81      public List<Circle> getFakeBalls() {
82          return getDecoyBalls();
83      }
84  
85      /**
86       * Returns copies of the active decoy balls, so callers cannot move the real ones.
87       *
88       * @return Copies of the decoy circles, or an empty list if the illusion is not active
89       */
90      @Override
91      public List<Circle> getDecoyBalls() {
92          List<Circle> copies = new ArrayList<>();
93          if (hasActiveIllusion()) {
94              for (Circle ball : fakeBalls) {
95                  copies.add(new Circle(ball.getX(), ball.getY(), ball.getRadius()));
96              }
97          }
98          return copies;
99      }
100 
101     /**
102      * Clears all active illusion decoy balls when the power shot ends.
103      */
104     @Override
105     protected void onShotFinished() {
106         this.illusionTicksRemaining = 0;
107         this.fakeBalls.clear();
108     }
109 
110     /**
111      * Checks whether illusion decoy balls are currently active.
112      *
113      * @return True if illusion is active
114      */
115     public boolean hasActiveIllusion() {
116         return illusionTicksRemaining > 0 && !fakeBalls.isEmpty();
117     }
118 
119     /**
120      * Returns the base power damage for Illusion.
121      *
122      * @return 30.0
123      */
124     @Override
125     public double getBaseDamage() {
126         return 30.0;
127     }
128 
129     /**
130      * Creates the CONFUSED effect applied on impact.
131      *
132      * @return ConfusedEffect instance
133      */
134     @Override
135     public Effect createEffect() {
136         return new ConfusedEffect();
137     }
138 
139     /**
140      * Updates decoy clone positions during each simulation tick.
141      *
142      * @param ball Real ball in play
143      */
144     @Override
145     protected void updateActiveShot(Ball ball){
146         updateClones(ball);
147     }
148 }
