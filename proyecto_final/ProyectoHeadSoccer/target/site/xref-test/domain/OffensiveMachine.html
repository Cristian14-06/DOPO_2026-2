3   import java.util.EnumSet;
4   import java.util.Set;
5   
6   /**
7    * Machine controller implementing the Ofensiva (offensive) profile prioritizing scoring.
8    * Pursues the ball across the pitch, positions behind it, jumps, kicks, and executes powers.
9    *
10   * @author Juan Espitia
11   * @author Cristian Salamanca
12   */
13  public class OffensiveMachine extends MachineController {
14      private static final long serialVersionUID = 1L;
15      private static final double ATTACK_OFFSET = 16.0;
16      private static final double BACKTRACK_OFFSET = 26.0;
17      private static final double MOVE_TOLERANCE = 4.0;
18      private static final double JUMP_MAX_DX = 32.0;
19      private static final double JUMP_HEAD_MARGIN_Y = 15.0;
20      private static final double KICK_MAX_DX = 30.0;
21      private static final double KICK_MAX_HEIGHT = 70.0;
22      private static final double KICK_DIRECTION_MARGIN = -12.0;
23      private static final double POWER_BALL_DISTANCE = 50.0;
24      private static final double FULL_POWER = 100.0;
25      private static final double CROSSBAR_BACKOFF = 45.0;
26  
27      /**
28       * Constructs an offensive machine controller for the given pitch side.
29       *
30       * @param side Pitch side controlled by this AI
31       */
32      public OffensiveMachine(Side side) {
33          super(side);
34      }
35  
36      /**
37       * Attacks the opponent's goal:
38       * <ul>
39       *   <li>If the ball lies on top of a crossbar, steps back so the ball can roll off.</li>
40       *   <li>Otherwise moves behind the ball relative to the opponent's goal
41       *       (a bit further back when the ball is already behind the player).</li>
42       *   <li>Jumps for headers when the ball is at head height or above.</li>
43       *   <li>Kicks when the ball is within reach and not behind the player.</li>
44       *   <li>Activates the power when the bar is full and the ball is close and in front.</li>
45       * </ul>
46       *
47       * @param match Read-only match view
48       * @param me Read-only view of the controlled player
49       * @param ball Read-only ball view
50       * @param rival Read-only view of the opponent
51       * @return Actions to perform during this tick
52       */
53      @Override
54      protected Set<PlayerAction> chooseActions(MatchView match, PlayerView me, BallView ball, PlayerView rival) {
55          Set<PlayerAction> actions = EnumSet.noneOf(PlayerAction.class);
56          double myX = me.getHeadShape().getX();
57          double bx = ball.getX();
58  
59          boolean ballBehindMe = (bx - myX) * side.dir() < 0;
60  
61          double targetX = ballBehindMe
62                  ? bx - side.dir() * BACKTRACK_OFFSET
63                  : bx - side.dir() * ATTACK_OFFSET;
64  
65          if (isBallOnCrossBar(match, ball)) {
66              moveToward(myX, bx - side.dir() * CROSSBAR_BACKOFF, MOVE_TOLERANCE, actions);
67              return actions;
68          }
69  
70          moveToward(myX, targetX, MOVE_TOLERANCE, actions);
71  
72          if (shouldJumpForBall(me, ball, JUMP_MAX_DX, JUMP_HEAD_MARGIN_Y)) {
73              actions.add(PlayerAction.JUMP);
74          }
75  
76          if (shouldKickBall(match, me, ball, KICK_MAX_DX, KICK_MAX_HEIGHT)
77                  && (bx - myX) * side.dir() > KICK_DIRECTION_MARGIN) {
78              actions.add(PlayerAction.KICK);
79          }
80  
81          if (me.getPowerProgress() >= FULL_POWER) {
82              double distBall = Math.hypot(myX - bx, me.getHeadShape().getY() - ball.getY());
83              if (distBall <= POWER_BALL_DISTANCE && (bx - myX) * side.dir() >= 0) {
84                  actions.add(PlayerAction.POWER);
85              }
86          }
87  
88          return actions;
89      }
90  
91      /*
92       * Checks whether the ball is lying on top of either crossbar, where it cannot be kicked.
93       */
94      private boolean isBallOnCrossBar(MatchView match, BallView ball) {
95          for (Side s : Side.values()) {
96              Rectangle bar = match.getCrossBar(s);
97              boolean above = ball.getY() < bar.getY();
98              boolean overBar = ball.getX() >= bar.getX() - ball.getRadius()
99                      && ball.getX() <= bar.getX() + bar.getWidth() + ball.getRadius();
100             if (above && overBar) return true;
101         }
102         return false;
103     }
