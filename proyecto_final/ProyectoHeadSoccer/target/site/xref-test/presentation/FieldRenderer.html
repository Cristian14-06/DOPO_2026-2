12  /**
13   * Renders the soccer pitch, stadium stands, turf lines, and both goals.
14   *
15   * @author Juan Espitia
16   * @author Cristian Salamanca
17   */
18  public class FieldRenderer {
19  
20      /**
21       * Draws the pitch background, turf markings, and left/right goals in field coordinates.
22       *
23       * @param g2d Scaled field graphics context
24       * @param match Read-only match view
25       */
26      public void draw(Graphics2D g2d, MatchView match) {
27          if (match == null) return;
28          int fieldW = match.getFieldWidth();
29          int fieldH = match.getFieldHeight();
30          int turfTop = fieldH - 40;
31          int bannerTop = turfTop - 12;
32  
33          g2d.setColor(new Color(62, 92, 128));
34          g2d.fillRect(0, 0, fieldW, bannerTop);
35  
36          g2d.setColor(new Color(74, 104, 140));
37          for (int row = 20; row < bannerTop - 8; row += 16) {
38              g2d.fillRect(0, row, fieldW, 6);
39          }
40  
41          g2d.setColor(new Color(30, 34, 42));
42          g2d.fillRect(0, bannerTop, fieldW, 12);
43          g2d.setColor(Color.WHITE);
44          g2d.setFont(new Font("Arial", Font.BOLD, 6));
45          g2d.drawString("DOPOSoccer 2026-2  •  HEAD SOCCER", 16, bannerTop + 8);
46          g2d.drawString("ESCUELA COLOMBIANA DE INGENIERÍA", 192, bannerTop + 8);
47  
48          g2d.setColor(new Color(46, 139, 87));
49          g2d.fillRect(0, turfTop, fieldW, fieldH - turfTop);
50  
51          g2d.setColor(new Color(56, 155, 98));
52          for (int i = 0; i < fieldW; i += 28) {
53              g2d.fillRect(i, turfTop, 14, fieldH - turfTop);
54          }
55  
56          g2d.setColor(Color.WHITE);
57          g2d.setStroke(new BasicStroke(1.0f));
58          g2d.drawLine(0, turfTop, fieldW, turfTop);
59          g2d.drawLine(0, fieldH - 1, fieldW, fieldH - 1);
60          g2d.drawLine(fieldW / 2, turfTop, fieldW / 2, fieldH);
61          g2d.drawOval(fieldW / 2 - 22, turfTop + 6, 44, 28);
62  
63          drawGoal(g2d, match, Side.LEFT);
64          drawGoal(g2d, match, Side.RIGHT);
65      }
66  
67      /*
68       * Renders the goal net, thin horizontal crossbar, and vertical goalpost for the specified side.
69       */
70      private void drawGoal(Graphics2D g2d, MatchView match, Side side) {
71          Rectangle area = match.getGoalArea(side);
72          Rectangle bar = match.getCrossBar(side);
73  
74          if (area != null) {
75              int rx = (int) area.getX();
76              int ry = (int) area.getY();
77              int rw = (int) area.getWidth();
78              int rh = (int) area.getHeight();
79  
80              g2d.setColor(new Color(245, 245, 245, 95));
81              g2d.fillRect(rx, ry, rw, rh);
82              g2d.setColor(new Color(210, 215, 225, 190));
83              g2d.setStroke(new BasicStroke(0.6f));
84              for (int x = rx; x <= rx + rw; x += 5) g2d.drawLine(x, ry, x, ry + rh);
85              for (int y = ry; y <= ry + rh; y += 6) g2d.drawLine(rx, y, rx + rw, y);
86  
87              int postW = 3;
88              int frontPostX = (side == Side.LEFT) ? (rx + rw - postW) : rx;
89              int backPostX = (side == Side.LEFT) ? rx : (rx + rw - 2);
90  
91              g2d.setColor(new Color(220, 225, 230));
92              g2d.fillRect(backPostX, ry, 2, rh);
93  
94              g2d.setColor(Color.WHITE);
95              g2d.fillRect(frontPostX, ry, postW, rh);
96              g2d.setColor(new Color(90, 95, 105));
97              g2d.setStroke(new BasicStroke(0.7f));
98              g2d.drawRect(frontPostX, ry, postW, rh);
99          }
100 
101         if (bar != null) {
102             int bx = (int) bar.getX();
103             int by = (int) bar.getY();
104             int bw = (int) bar.getWidth();
105             int bh = (int) bar.getHeight();
106 
107             g2d.setColor(Color.WHITE);
108             g2d.fillRect(bx, by, bw, bh);
109             g2d.setColor(new Color(90, 95, 105));
110             g2d.setStroke(new BasicStroke(0.8f));
111             g2d.drawRect(bx, by, bw, bh);
112         }
113     }
