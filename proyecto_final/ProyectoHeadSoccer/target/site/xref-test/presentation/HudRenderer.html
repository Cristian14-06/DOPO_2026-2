15  /**
16   * Renders the top HUD including scoreboard, timer or goal target, and player bars.
17   *
18   * @author Juan Espitia
19   * @author Cristian Salamanca
20   */
21  public class HudRenderer {
22      private final ImageAssets assets;
23  
24      /**
25       * Constructs a HUD renderer using shared image assets for country flags.
26       *
27       * @param assets Loaded image assets
28       */
29      public HudRenderer(ImageAssets assets) {
30          this.assets = assets;
31      }
32  
33      /**
34       * Draws the scoreboard, match mode/timer, and both player HUD panels.
35       *
36       * @param g2d Screen-space graphics context
37       * @param match Read-only match view
38       * @param setup Match configuration holding goal target and machine profile
39       * @param panelWidth Current width of the match canvas in pixels
40       */
41      public void draw(Graphics2D g2d, MatchView match, MatchSetup setup, int panelWidth) {
42          if (match == null) return;
43  
44          g2d.setColor(new Color(15, 20, 25, 230));
45          g2d.fillRoundRect(panelWidth / 2 - 160, 10, 320, 74, 14, 14);
46          g2d.setColor(new Color(245, 195, 35));
47          g2d.setStroke(new BasicStroke(1.5f));
48          g2d.drawRoundRect(panelWidth / 2 - 160, 10, 320, 74, 14, 14);
49  
50          g2d.setFont(new Font("Arial", Font.BOLD, 11));
51          g2d.setColor(new Color(180, 220, 255));
52          String modeTag = (match.getModality() == Modality.PVC)
53                  ? "PvM • " + setup.getMachineProfile().getDisplayName()
54                  : "PvP";
55          g2d.drawString(modeTag, panelWidth / 2 - g2d.getFontMetrics().stringWidth(modeTag) / 2, 23);
56  
57          int leftGoals = match.getGoals(Side.LEFT);
58          int rightGoals = match.getGoals(Side.RIGHT);
59          g2d.setFont(new Font("Impact", Font.PLAIN, 36));
60          g2d.setColor(Color.WHITE);
61          String scoreText = leftGoals + "  -  " + rightGoals;
62          FontMetrics fm = g2d.getFontMetrics();
63          g2d.drawString(scoreText, panelWidth / 2 - fm.stringWidth(scoreText) / 2, 48);
64  
65          g2d.setFont(new Font("Monospaced", Font.BOLD, 14));
66          g2d.setColor(Color.GREEN);
67          String centerInfo;
68          if (match.isSuddenDeath()) {
69              centerInfo = "GOL DE ORO";
70              g2d.setColor(Color.YELLOW);
71          } else if (match.getRemainingSeconds() < 0) {
72              centerInfo = "META: " + setup.getTargetGoals() + " GOLES";
73          } else {
74              int sec = match.getRemainingSeconds();
75              centerInfo = String.format("%02d:%02d", sec / 60, sec % 60);
76          }
77          g2d.drawString(centerInfo, panelWidth / 2 - g2d.getFontMetrics().stringWidth(centerInfo) / 2, 68);
78  
79          PlayerView p1 = match.getPlayer(Side.LEFT);
80          if (p1 != null) {
81              drawPlayerHUD(g2d, p1, 25, 14, 220);
82          }
83  
84          PlayerView p2 = match.getPlayer(Side.RIGHT);
85          if (p2 != null) {
86              drawPlayerHUD(g2d, p2, panelWidth - 245, 14, 220);
87          }
88      }
89  
90      /*
91       * Renders a single player's name, country flag, health bar, and power gauge in the HUD.
92       */
93      private void drawPlayerHUD(Graphics2D g2d, PlayerView p, int x, int y, int barW) {
94          int textX = x;
95          BufferedImage flagImg = (assets != null) ? assets.getFlag(p.getCountry()) : null;
96          if (flagImg != null) {
97              g2d.drawImage(flagImg, x, y + 1, 20, 13, null);
98              textX = x + 25;
99          }
100         g2d.setFont(new Font("Arial", Font.BOLD, 13));
101         g2d.setColor(Color.WHITE);
102         String name = p.getName() + " (" + p.getCountry().getDisplayName() + ")";
103         g2d.drawString(name, textX, y + 12);
104 
105         int hpY = y + 18;
106         g2d.setColor(Color.BLACK);
107         g2d.fillRect(x, hpY, barW, 14);
108 
109         double hpRatio = Math.max(0.0, Math.min(1.0, p.getLife() / p.getMaxLife()));
110         g2d.setColor(healthColor(hpRatio));
111         g2d.fillRect(x + 1, hpY + 1, (int) ((barW - 2) * hpRatio), 12);
112         g2d.setColor(Color.WHITE);
113         g2d.drawRect(x, hpY, barW, 14);
114 
115         g2d.setFont(new Font("Arial", Font.BOLD, 10));
116         String hpText = (int) p.getLife() + "/" + (int) p.getMaxLife() + " HP";
117         g2d.drawString(hpText, x + barW / 2 - g2d.getFontMetrics().stringWidth(hpText) / 2, hpY + 11);
118 
119         int powY = hpY + 16;
120         g2d.setColor(Color.BLACK);
121         g2d.fillRect(x, powY, barW, 10);
122 
123         double powRatio = Math.max(0.0, Math.min(1.0, p.getPowerProgress() / 100.0));
124         g2d.setColor(powRatio >= 1.0 ? Color.YELLOW : new Color(240, 140, 0));
125         g2d.fillRect(x + 1, powY + 1, (int) ((barW - 2) * powRatio), 8);
126         g2d.setColor(Color.GRAY);
127         g2d.drawRect(x, powY, barW, 10);
128 
129         if (p.hasAura()) {
130             g2d.setFont(new Font("Arial", Font.BOLD, 9));
131             g2d.setColor(Color.YELLOW);
132             g2d.drawString("⚡ AURA ACTIVA ⚡", x + barW / 2 - 40, powY + 9);
133         }
134     }
135 
136     /*
137      * Chooses the health bar color: green above half life, orange above a quarter, red below.
138      */
139     private Color healthColor(double hpRatio) {
140         if (hpRatio > 0.5) {
141             return new Color(40, 200, 40);
142         }
143         if (hpRatio > 0.25) {
144             return Color.ORANGE;
145         }
146         return Color.RED;
147     }
