6   /**
7    * Main menu screen built with standard Swing components.
8    *
9    * @author Juan Espitia

12  public class MenuPanel extends JPanel {
13  
14      /**
15       * Constructs the main menu panel with New Game and Open Match actions.
16       *
17       * @param onNewGame Action invoked when the user clicks New Game
18       * @param onOpenMatch Action invoked when the user clicks Open Match
19       */
20      public MenuPanel(Runnable onNewGame, Runnable onOpenMatch) {
21          setLayout(new GridBagLayout());
22          setBackground(new Color(22, 34, 52));
23  
24          JPanel content = new JPanel();
25          content.setOpaque(false);
26          content.setLayout(new BoxLayout(content, BoxLayout.Y_AXIS));
27  
28          JLabel title = new JLabel("HEAD SOCCER GAME");
29          title.setFont(new Font("Impact", Font.BOLD, 48));
30          title.setForeground(new Color(255, 220, 40));
31          title.setAlignmentX(CENTER_ALIGNMENT);
32  
33          JLabel subtitle = new JLabel("DOPOSoccer 2026-2  •  Proyecto Final POOB");
34          subtitle.setFont(new Font("Arial", Font.BOLD, 16));
35          subtitle.setForeground(Color.WHITE);
36          subtitle.setAlignmentX(CENTER_ALIGNMENT);
37  
38          JPanel buttonRow = new JPanel(new FlowLayout(FlowLayout.CENTER, 24, 0));
39          buttonRow.setOpaque(false);
40  
41          JButton btnNewGame = Theme.createStyledButton("NEW GAME", Theme.ACCENT_GOLD, Color.BLACK, 20);
42          btnNewGame.addActionListener(e -> onNewGame.run());
43  
44          JButton btnOpen = Theme.createStyledButton("ABRIR PARTIDA", Theme.ACTION_BLUE, Color.WHITE, 18);
45          btnOpen.addActionListener(e -> onOpenMatch.run());
46  
47          buttonRow.add(btnNewGame);
48          buttonRow.add(btnOpen);
49  
50          JPanel controlsBox = new JPanel();
51          controlsBox.setLayout(new BoxLayout(controlsBox, BoxLayout.Y_AXIS));
52          controlsBox.setBackground(new Color(12, 18, 28, 210));
53          controlsBox.setBorder(BorderFactory.createCompoundBorder(
54                  BorderFactory.createLineBorder(new Color(245, 195, 35), 2),
55                  BorderFactory.createEmptyBorder(14, 24, 14, 24)
56          ));
57          controlsBox.setAlignmentX(CENTER_ALIGNMENT);
58  
59          JLabel ctrlTitle = new JLabel("CONTROLES DE JUEGO (Mismo Teclado):");
60          ctrlTitle.setFont(new Font("Arial", Font.BOLD, 15));
61          ctrlTitle.setForeground(Color.YELLOW);
62  
63          JLabel p1Ctrl = new JLabel("• Jugador 1: A / D (Mover), W (Saltar), S (Patear), Q (Activar Poder)");
64          p1Ctrl.setFont(new Font("Arial", Font.PLAIN, 14));
65          p1Ctrl.setForeground(Color.WHITE);
66  
67          JLabel p2Ctrl = new JLabel("• Jugador 2: ← / → (Mover), ↑ (Saltar), ↓ (Patear), Ctrl Der (Activar Poder)");
68          p2Ctrl.setFont(new Font("Arial", Font.PLAIN, 14));
69          p2Ctrl.setForeground(Color.WHITE);
70  
71          controlsBox.add(ctrlTitle);
72          controlsBox.add(Box.createVerticalStrut(8));
73          controlsBox.add(p1Ctrl);
74          controlsBox.add(Box.createVerticalStrut(6));
75          controlsBox.add(p2Ctrl);
76  
77          content.add(title);
78          content.add(Box.createVerticalStrut(10));
79          content.add(subtitle);
80          content.add(Box.createVerticalStrut(36));
81          content.add(buttonRow);
82          content.add(Box.createVerticalStrut(32));
83          content.add(controlsBox);
84  
85          add(content);
86      }
87  
88      /**
89       * Paints a stadium-style backdrop behind the Swing menu controls.
90       *
91       * @param g Graphics context
92       */
93      @Override
94      protected void paintComponent(Graphics g) {
95          super.paintComponent(g);
96          int w = getWidth();
97          int h = getHeight();
98          g.setColor(new Color(45, 72, 105));
99          g.fillRect(0, 0, w, (int) (h * 0.72));
100         g.setColor(new Color(30, 34, 40));
101         g.fillRect(0, (int) (h * 0.72), w, (int) (h * 0.04));
102         g.setColor(new Color(38, 120, 74));
103         g.fillRect(0, (int) (h * 0.76), w, h - (int) (h * 0.76));
104         g.setColor(new Color(0, 0, 0, 120));
105         g.fillRect(0, 0, w, h);
106     }
107 }
