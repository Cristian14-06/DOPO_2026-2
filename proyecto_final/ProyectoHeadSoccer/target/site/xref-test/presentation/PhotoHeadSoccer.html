93      /**
94       * Builds a MatchConfig from the current UI selections and starts a new match.
95       */
96      public void startConfiguredMatch() {
97          try {
98              MatchConfig config = setup.toConfig();
99              this.lastScorerText = "¡Anotación!";
100             this.winnerNotification = "Fin del juego";
101             game.newMatch(config);
102             parentFrame.setCurrentScreen(Screen.MATCH);

104             Log.record(ex);
105             JOptionPane.showMessageDialog(this, ex.getMessage(), "Error de configuración", JOptionPane.ERROR_MESSAGE);
106         }

109     /**
110      * Updates the goal banner text with the scoring player's name.
111      *
112      * @param scorer Side of the player who scored
113      */
114     @Override
115     public void onGoal(Side scorer) {
116         soundPlayer.play(Sound.GOAL);
117         MatchView match = game.getMatch();
118         if (match != null && match.getPlayer(scorer) != null) {
119             lastScorerText = "¡GOL DE " + match.getPlayer(scorer).getName().toUpperCase() + "!";
120         } else {
121             lastScorerText = "¡GOL!";
122         }
123     }

136     /**
137      * Updates the end-of-match notification message with the winner and awarded points.
138      *
139      * @param winner Winning side, or null if tied
140      * @param winnerPoints Points awarded to the winner
141      * @param loserPoints Points awarded to the loser
142      */
143     @Override
144     public void onMatchFinished(Side winner, int winnerPoints, int loserPoints) {
145         soundPlayer.play(Sound.FINISH);
146         if (winner == null) {
147             winnerNotification = "¡EMPATE TÉCNICO! (+" + winnerPoints + " PUNTOS)";
148             return;
149         }
150         MatchView match = game.getMatch();
151         String name = (match != null && match.getPlayer(winner) != null)
152                 ? match.getPlayer(winner).getName().toUpperCase()
153                 : winner.name();
154         winnerNotification = "¡GANADOR: " + name + "! (+" + winnerPoints + " PUNTOS)";
155     }

214     /**
215      * Renders the active application screen onto the panel.
216      *
217      * @param g Swing graphics context
218      */
219     @Override
220     protected void paintComponent(Graphics g) {
221         super.paintComponent(g);
222         Graphics2D g2d = (Graphics2D) g;
223         g2d.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
224 
225         MatchView match = game.getMatch();
226         if (match != null) {
227             drawMatchScreen(g2d, match);
228         }
229     }

231     /*
232      * Renders the active match screen, scaled pitch entities, HUD, and state overlays.
233      */
234     private void drawMatchScreen(Graphics2D g2d, MatchView match) {
235         if (match == null) return;
236         int w = getWidth();
237         int h = getHeight();
238         int fieldW = match.getFieldWidth();
239         int fieldH = match.getFieldHeight();
240 
241         g2d.setColor(new Color(48, 72, 104));
242         g2d.fillRect(0, 0, w, h);
243 
244         int hudReserve = (int) (h * 0.16);
245         double scaleX = (double) w / fieldW;
246         double scaleY = (double) (h - hudReserve) / fieldH;
247         double currentScale = Math.min(scaleX, scaleY);
248 
249         int offsetX = (int) ((w - fieldW * currentScale) / 2.0);
250         int offsetY = (int) (h - fieldH * currentScale);
251 
252         Graphics2D gField = (Graphics2D) g2d.create();
253         gField.translate(offsetX, offsetY);
254         gField.scale(currentScale, currentScale);
255 
256         fieldRenderer.draw(gField, match);
257 
258         PlayerView left = match.getPlayer(Side.LEFT);
259         PlayerView right = match.getPlayer(Side.RIGHT);
260         if (left != null) playerRenderer.draw(gField, left);
261         if (right != null) playerRenderer.draw(gField, right);
262         ballRenderer.draw(gField, match);
263 
264         gField.dispose();
265 
266         hudRenderer.draw(g2d, match, setup, w);
267         overlayRenderer.draw(g2d, match, w, h, lastScorerText, winnerNotification);
268     }

270     /**
271      * Dispatches mouse click events across match overlay buttons.
272      *
273      * @param e Mouse click event
274      */
275     public void handleMouseClick(MouseEvent e) {
276         Point p = e.getPoint();
277         Screen screen = parentFrame.getCurrentScreen();
278 
279         if (screen == Screen.MATCH) {
280             MatchView match = game.getMatch();
281             if (match != null && match.getStatus() == GameStatus.PAUSED && btnSaveGame.contains(p)) {
282                 saveCurrentMatch();
283                 repaint();
284             } else if (match != null && match.getStatus() == GameStatus.FINISHED && btnUpgrades.contains(p)) {
285                 parentFrame.setCurrentScreen(Screen.UPGRADES);
286                 repaint();
287             } else if (match != null && match.getStatus() == GameStatus.FINISHED && btnMenu.contains(p)) {
288                 parentFrame.setCurrentScreen(Screen.MENU);
289                 repaint();
290             }
291         }
292     }

294     /*
295      * Opens a file chooser dialog to save the currently paused match.
296      */
297     private void saveCurrentMatch() {
298         JFileChooser chooser = new JFileChooser();
299         chooser.setDialogTitle("Guardar partida de DOPOSoccer");
300         chooser.setFileFilter(new FileNameExtensionFilter("Partidas DOPOSoccer (*.dat)", "dat"));
301         if (chooser.showSaveDialog(this) == JFileChooser.APPROVE_OPTION) {
302             File file = chooser.getSelectedFile();
303             if (!file.getName().toLowerCase().endsWith(".dat")) {
304                 file = new File(file.getParentFile(), file.getName() + ".dat");
305             }
306             try {
307                 game.save(file);
308                 JOptionPane.showMessageDialog(this, "Partida guardada correctamente.", "Guardar", JOptionPane.INFORMATION_MESSAGE);
309             } catch (DOPOSoccerException ex) {
310                 Log.record(ex);
311                 JOptionPane.showMessageDialog(this, ex.getMessage(), "Error al guardar", JOptionPane.ERROR_MESSAGE);
312             }
313         }
314     }

316     /**
317      * Opens a file chooser dialog to load a previously saved match.
318      */
319     public void openSavedMatch() {
320         JFileChooser chooser = new JFileChooser();
321         chooser.setDialogTitle("Abrir partida de DOPOSoccer");
322         chooser.setFileFilter(new FileNameExtensionFilter("Partidas DOPOSoccer (*.dat)", "dat"));
323         if (chooser.showOpenDialog(this) == JFileChooser.APPROVE_OPTION) {
324             try {
325                 game.open(chooser.getSelectedFile());
326                 this.lastScorerText = "¡Anotación!";
327                 this.winnerNotification = "Fin del juego";
328                 parentFrame.setCurrentScreen(Screen.MATCH);
329             } catch (DOPOSoccerException ex) {
330                 Log.record(ex);
331                 JOptionPane.showMessageDialog(this, ex.getMessage(), "Error al abrir", JOptionPane.ERROR_MESSAGE);
332             }
333         }
