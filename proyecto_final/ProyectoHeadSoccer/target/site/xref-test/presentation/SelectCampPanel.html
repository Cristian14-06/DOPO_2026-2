3   import javax.swing.*;
4   import java.awt.*;
5   
6   /**
7    * Field resolution selection screen built with standard Swing buttons.
8    *
9    * @author Juan Espitia
10   * @author Cristian Salamanca
11   */
12  public class SelectCampPanel extends JPanel {
13      private final MatchSetup setup;
14      private final JButton btnSmall;
15      private final JButton btnMedium;
16      private final JButton btnLarge;
17  
18      /**
19       * Constructs the resolution selection screen.
20       *
21       * @param photo Match panel holding the selected resolution
22       * @param onResolutionChanged Callback invoked when a resolution button is clicked
23       * @param onBack Action invoked to return to player selection
24       * @param onStart Action invoked to launch the configured match
25       */
26      public SelectCampPanel(
27              MatchSetup setup,
28              java.util.function.Consumer<FieldResolution> onResolutionChanged,
29              Runnable onBack,
30              Runnable onStart
31      ) {
32          this.setup = setup;
33          setLayout(new GridBagLayout());
34          setBackground(new Color(22, 34, 52));
35  
36          JPanel card = new JPanel();
37          card.setLayout(new BoxLayout(card, BoxLayout.Y_AXIS));
38          card.setBackground(new Color(25, 30, 42, 235));
39          card.setBorder(BorderFactory.createCompoundBorder(
40                  BorderFactory.createLineBorder(new Color(255, 200, 50), 2),
41                  BorderFactory.createEmptyBorder(24, 34, 24, 34)
42          ));
43  
44          JLabel title = new JLabel("SELECT THE CAMP");
45          title.setFont(new Font("Impact", Font.BOLD, 38));
46          title.setForeground(Color.YELLOW);
47          title.setAlignmentX(CENTER_ALIGNMENT);
48  
49          JLabel subtitle = new JLabel("Tamaño de ventana / resolución:");
50          subtitle.setFont(new Font("Arial", Font.BOLD, 16));
51          subtitle.setForeground(Color.WHITE);
52          subtitle.setAlignmentX(CENTER_ALIGNMENT);
53  
54          JPanel resRow = new JPanel(new FlowLayout(FlowLayout.CENTER, 14, 6));
55          resRow.setOpaque(false);
56  
57          btnSmall = createResButton("Pequeño (700×400)");
58          btnMedium = createResButton("Mediano (1050×600)");
59          btnLarge = createResButton("Grande (1400×800)");
60  
61          btnSmall.addActionListener(e -> {
62              setup.setResolution(FieldResolution.SMALL);
63              onResolutionChanged.accept(FieldResolution.SMALL);
64              refreshView();
65          });
66          btnMedium.addActionListener(e -> {
67              setup.setResolution(FieldResolution.MEDIUM);
68              onResolutionChanged.accept(FieldResolution.MEDIUM);
69              refreshView();
70          });
71          btnLarge.addActionListener(e -> {
72              setup.setResolution(FieldResolution.LARGE);
73              onResolutionChanged.accept(FieldResolution.LARGE);
74              refreshView();
75          });
76  
77          resRow.add(btnSmall);
78          resRow.add(btnMedium);
79          resRow.add(btnLarge);
80  
81          JPanel navRow = new JPanel(new FlowLayout(FlowLayout.CENTER, 18, 0));
82          navRow.setOpaque(false);
83  
84          JButton btnBack = new JButton("← VOLVER");
85          btnBack.setFont(new Font("Arial", Font.BOLD, 15));
86          btnBack.setBackground(new Color(80, 90, 105));
87          btnBack.setForeground(Color.WHITE);
88          btnBack.setFocusPainted(false);
89          btnBack.setPreferredSize(new Dimension(150, 48));
90          btnBack.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
91          btnBack.addActionListener(e -> onBack.run());
92  
93          JButton btnStart = new JButton("⚽ START ⚽");
94          btnStart.setFont(new Font("Arial", Font.BOLD, 20));
95          btnStart.setBackground(new Color(40, 180, 40));
96          btnStart.setForeground(Color.WHITE);
97          btnStart.setFocusPainted(false);
98          btnStart.setPreferredSize(new Dimension(220, 48));
99          btnStart.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
100         btnStart.addActionListener(e -> onStart.run());
101 
102         navRow.add(btnBack);
103         navRow.add(btnStart);
104 
105         card.add(title);
106         card.add(Box.createVerticalStrut(18));
107         card.add(subtitle);
108         card.add(Box.createVerticalStrut(12));
109         card.add(resRow);
110         card.add(Box.createVerticalStrut(26));
111         card.add(navRow);
112 
113         add(card);
114         refreshView();
115     }
116 
117     /**
118      * Highlights the currently selected field resolution button.
119      */
120     public final void refreshView() {
121         FieldResolution res = setup.getResolution();
122         Theme.styleToggle(btnSmall, res == FieldResolution.SMALL);
123         Theme.styleToggle(btnMedium, res == FieldResolution.MEDIUM);
124         Theme.styleToggle(btnLarge, res == FieldResolution.LARGE);
125     }
126 
127     /*
128      * Creates a resolution toggle button with uniform typography and hand cursor.
129      */
130     private JButton createResButton(String text) {
131         JButton btn = new JButton(text);
132         btn.setFont(new Font("Arial", Font.BOLD, 14));
133         btn.setFocusPainted(false);
134         btn.setPreferredSize(new Dimension(175, 42));
135         btn.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
136         return btn;
137     }
138 }
