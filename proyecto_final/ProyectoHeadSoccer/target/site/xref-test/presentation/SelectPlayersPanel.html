196     /**
197      * Synchronizes button states and labels with the current configuration in {@link PhotoHeadSoccer}.
198      */
199     public final void refreshView() {
200         boolean vsMachine = setup.getModality() == Modality.PVC;
201         Theme.styleToggle(btnPvP, !vsMachine);
202         Theme.styleToggle(btnPvM, vsMachine);
203 
204         lblProfileTitle.setVisible(vsMachine);
205         btnNextProfile.setVisible(vsMachine);
206         MachineProfile profile = setup.getMachineProfile();
207         btnNextProfile.setText(profile.getDisplayName() + " ↻");
208         Theme.styleToggle(btnNextProfile, true);
209 
210         Country left = setup.getLeftCountry();
211         Country right = setup.getRightCountry();
212         lblP1Country.setText("País: " + left.getDisplayName());
213         lblP1Power.setText("Poder: " + left.getPowerName());
214 
215         lblP2Title.setText(vsMachine ? "MÁQUINA (Derecha)" : "JUGADOR 2 (Derecha)");
216         txtP2Name.setEnabled(!vsMachine);
217         txtP2Name.setText(vsMachine ? ("COM " + profile.getDisplayName()) : setup.getRightPlayerName());
218         lblP2Country.setText("País: " + right.getDisplayName());
219         lblP2Power.setText("Poder: " + right.getPowerName());
220 
221         btnTypeTime.setText("Por Tiempo (" + setup.getMatchDurationMinutes() + " min) ↻");
222         btnTypeGoals.setText("Por Goles (" + setup.getTargetGoals() + " goles) ↻");
223         Theme.styleToggle(btnTypeTime, setup.getMatchType() == MatchType.TIME);
224         Theme.styleToggle(btnTypeGoals, setup.getMatchType() == MatchType.GOALS);
225     }

227     /*
228      * Builds a styled vertical card container for player configuration fields.
229      */
230     private JPanel createPlayerCard(Color borderColor) {
231         JPanel panel = new JPanel();
232         panel.setLayout(new BoxLayout(panel, BoxLayout.Y_AXIS));
233         panel.setBackground(new Color(40, 46, 62));
234         panel.setBorder(BorderFactory.createCompoundBorder(
235                 BorderFactory.createLineBorder(borderColor, 2),
236                 BorderFactory.createEmptyBorder(14, 18, 14, 18)
237         ));
238         return panel;
239     }

241     /*
242      * Creates a styled text label with the given font size, weight, and color.
243      */
244     private JLabel createLabel(String text, int size, boolean bold, Color color) {
245         JLabel lbl = new JLabel(text);
246         lbl.setFont(new Font("Arial", bold ? Font.BOLD : Font.PLAIN, size));
247         lbl.setForeground(color);
248         return lbl;
249     }
250 
251     /*
252      * Creates a toggleable configuration option button.
253      */
254     private JButton createOptionButton(String text) {
255         JButton btn = new JButton(text);
256         btn.setFont(new Font("Arial", Font.BOLD, 13));
257         btn.setFocusPainted(false);
258         btn.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
259         return btn;
260     }
261 
262     /*
263      * Creates a colored action button for navigation or screen transitions.
264      */
265     private JButton createActionButton(String text, Color bg, Color fg) {
266         JButton btn = new JButton(text);
267         btn.setFont(new Font("Arial", Font.BOLD, 14));
268         btn.setBackground(bg);
269         btn.setForeground(fg);
270         btn.setFocusPainted(false);
271         btn.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
272         return btn;
273     }

275     /*
276      * Copies non-blank player names from the text fields into the shared configuration holder.
277      */
278     private void savePlayerNames() {
279         String n1 = txtP1Name.getText().trim();
280         if (!n1.isEmpty()) setup.setLeftPlayerName(n1);
281         if (setup.getModality() == Modality.PVP) {
282             String n2 = txtP2Name.getText().trim();
283             if (!n2.isEmpty()) setup.setRightPlayerName(n2);
284         }
285     }
