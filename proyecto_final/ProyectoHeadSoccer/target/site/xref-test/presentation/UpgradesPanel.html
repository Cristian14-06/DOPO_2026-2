13  /**
14   * Screen allowing players to spend accumulated points to upgrade their statistics.
15   *
16   * @author Juan Espitia
17   * @author Cristian Salamanca
18   */
19  public class UpgradesPanel extends JPanel {
20      private final DOPOSoccer game;
21      private final MatchSetup setup;
22      private final Runnable onBack;
23      private final JLabel title;
24      private final JPanel playersContainer;
25      private final JButton btnBack;
26  
27      /**
28       * Constructs the upgrades panel layout and back navigation button.
29       *
30       * @param game Domain facade instance
31       * @param photo Match panel holding configured player names and modality
32       * @param onBack Callback executed when the user clicks Back
33       */
34      public UpgradesPanel(DOPOSoccer game, MatchSetup setup, Runnable onBack) {
35          this.game = game;
36          this.setup = setup;
37          this.onBack = onBack;
38  
39          setLayout(new GridBagLayout());
40          setBackground(new Color(22, 34, 52));
41  
42          this.title = new JLabel("MEJORAR ESTADÍSTICAS");
43          this.title.setFont(new Font("Impact", Font.BOLD, 34));
44          this.title.setForeground(Color.YELLOW);
45          this.title.setAlignmentX(CENTER_ALIGNMENT);
46  
47          this.playersContainer = new JPanel(new GridLayout(1, 0, 20, 0));
48          this.playersContainer.setOpaque(false);
49  
50          this.btnBack = new JButton("← VOLVER");
51          this.btnBack.setFont(new Font("Arial", Font.BOLD, 15));
52          this.btnBack.setBackground(new Color(80, 90, 105));
53          this.btnBack.setForeground(Color.WHITE);
54          this.btnBack.setFocusPainted(false);
55          this.btnBack.setAlignmentX(CENTER_ALIGNMENT);
56          this.btnBack.addActionListener(e -> this.onBack.run());
57  
58          JPanel card = new JPanel();
59          card.setLayout(new BoxLayout(card, BoxLayout.Y_AXIS));
60          card.setBackground(new Color(25, 30, 42, 235));
61          card.setBorder(BorderFactory.createCompoundBorder(
62                  BorderFactory.createLineBorder(new Color(255, 200, 50), 2),
63                  BorderFactory.createEmptyBorder(20, 28, 20, 28)
64          ));
65          card.add(title);
66          card.add(Box.createVerticalStrut(16));
67          card.add(playersContainer);
68          card.add(Box.createVerticalStrut(20));
69          card.add(btnBack);
70  
71          add(card);
72      }
73  
74      /**
75       * Rebuilds the player upgrade cards with the latest profiles and points.
76       */
77      public void refreshView() {
78          playersContainer.removeAll();
79          playersContainer.add(buildPlayerCard(setup.getLeftPlayerName()));
80          if (setup.getModality() == Modality.PVP) {
81              playersContainer.add(buildPlayerCard(setup.getRightPlayerName()));
82          }
83          playersContainer.revalidate();
84          playersContainer.repaint();
85      }
86  
87      /*
88       * Builds an upgrade card showing the player's points and stat upgrade controls.
89       */
90      private JPanel buildPlayerCard(String playerName) {
91          JPanel card = new JPanel();
92          card.setLayout(new BoxLayout(card, BoxLayout.Y_AXIS));
93          card.setBackground(new Color(40, 46, 62));
94          card.setBorder(BorderFactory.createCompoundBorder(
95                  BorderFactory.createLineBorder(new Color(100, 150, 255), 2),
96                  BorderFactory.createEmptyBorder(14, 18, 14, 18)
97          ));
98  
99          try {
100             ProfileView profile = game.getProfile(playerName);
101             JLabel lblName = new JLabel(profile.getName());
102             lblName.setFont(new Font("Arial", Font.BOLD, 18));
103             lblName.setForeground(Color.WHITE);
104 
105             JLabel lblPoints = new JLabel("Puntos: " + profile.getPoints());
106             lblPoints.setFont(new Font("Arial", Font.BOLD, 15));
107             lblPoints.setForeground(new Color(255, 220, 80));
108 
109             card.add(lblName);
110             card.add(Box.createVerticalStrut(6));
111             card.add(lblPoints);
112             card.add(Box.createVerticalStrut(12));
113 
114             for (StatType type : StatType.values()) {
115                 JPanel row = new JPanel(new BorderLayout(12, 0));
116                 row.setOpaque(false);
117                 JLabel statName = new JLabel(type.name() + ": " + profile.getStat(type) + "/10");
118                 statName.setForeground(Color.WHITE);
119                 statName.setFont(new Font("Arial", Font.PLAIN, 14));
120                 int cost = profile.upgradeCost(type);
121                 JButton upgrade = new JButton((cost > 0) ? "+ (" + cost + " pts)" : "MÁX");
122                 upgrade.setFocusPainted(false);
123                 if (cost == 0) {
124                     upgrade.setEnabled(false);
125                 } else {
126                     upgrade.addActionListener(e -> handleUpgrade(playerName, type));
127                 }
128                 row.add(statName, BorderLayout.WEST);
129                 row.add(upgrade, BorderLayout.EAST);
130                 card.add(row);
131                 card.add(Box.createVerticalStrut(8));
132             }
133         } catch (DOPOSoccerException e) {
134             Log.record(e);
135             JOptionPane.showMessageDialog(this, e.getMessage(), "Error de perfil", JOptionPane.ERROR_MESSAGE);
136         }
137 
138         return card;
139     }
140 
141     /*
142      * Requests a stat upgrade through the domain facade and refreshes the panel on success.
143      */
144     private void handleUpgrade(String playerName, StatType stat) {
145         try {
146             game.upgrade(playerName, stat);
147             refreshView();
148         } catch (DOPOSoccerException ex) {
149             Log.record(ex);
150             JOptionPane.showMessageDialog(this, ex.getMessage(), "Mejora", JOptionPane.WARNING_MESSAGE);
151         }
152     }
