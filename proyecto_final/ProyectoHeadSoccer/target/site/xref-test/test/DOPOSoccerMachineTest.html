59      /**
60       * Verifies that DefensiveMachine returns to its home zone when the ball is far,
61       * and intercepts, jumps, and kicks when the ball enters its defensive half.
62       */
63      @Test
64      public void testDefensiveMachineReturnsHomeWhenBallFarAndDefendsWhenNear() {
65          DefensiveMachine machine = new DefensiveMachine(Side.RIGHT);
66  
67          StubMatch farMatch = new StubMatch()
68                  .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 200, Field.HEIGHT - 38))
69                  .withBall(new StubBall(80, Field.HEIGHT - 8, 0, 0, false));
70          Set<PlayerAction> farActions = machine.decide(farMatch);
71          assertTrue("Defensive machine should move right toward home position when ball is far",
72                  farActions.contains(PlayerAction.RIGHT));
73          assertFalse(farActions.contains(PlayerAction.KICK));
74  
75          StubMatch nearLowMatch = new StubMatch()
76                  .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 260, Field.HEIGHT - 38))
77                  .withBall(new StubBall(245, Field.HEIGHT - 12, -3, 0, false));
78          Set<PlayerAction> nearLowActions = machine.decide(nearLowMatch);
79          assertTrue("Defensive machine should kick a low ball in front of it",
80                  nearLowActions.contains(PlayerAction.KICK));
81  
82          StubMatch nearHighMatch = new StubMatch()
83                  .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 270, Field.HEIGHT - 38))
84                  .withBall(new StubBall(265, Field.HEIGHT - 60, -2, 0, false));
85          Set<PlayerAction> nearHighActions = machine.decide(nearHighMatch);
86          assertTrue("Defensive machine should jump for a high ball in its zone",
87                  nearHighActions.contains(PlayerAction.JUMP));
88      }

90      /**
91       * Verifies that ClumsyMachine is deterministic with a fixed seed, holds each chosen action
92       * for at least 12 ticks, and activates power when the gauge is full.
93       */
94      @Test
95      public void testClumsyMachineDeterministicAndHoldsActions() {
96          MachineController m1 = MachineProfile.TORPE.create(Side.RIGHT, new Random(42L));
97          MachineController m2 = MachineProfile.TORPE.create(Side.RIGHT, new Random(42L));
98  
99          StubPlayer me = new StubPlayer(Side.RIGHT, 250, Field.HEIGHT - 38).withPower(100.0);
100         StubMatch match = new StubMatch()
101                 .withPlayer(Side.RIGHT, me)
102                 .withBall(new StubBall(175, Field.HEIGHT - 8, 0, 0, false));
103 
104         Set<PlayerAction> first = m1.decide(match);
105         assertEquals(first, m2.decide(match));
106         for (int t = 1; t < 12; t++) {
107             assertEquals("ClumsyMachine must hold its buffered action for at least 12 ticks",
108                     first, m1.decide(match));
109             m2.decide(match);
110         }
111 
112         boolean sawPower = false;
113         boolean sawMove = false;
114         for (int t = 0; t < 1000; t++) {
115             Set<PlayerAction> actions = m1.decide(match);
116             if (actions.contains(PlayerAction.POWER)) sawPower = true;
117             if (actions.contains(PlayerAction.LEFT) || actions.contains(PlayerAction.RIGHT)) sawMove = true;
118         }
119         assertTrue("ClumsyMachine should eventually activate full power", sawPower);
120         assertTrue("ClumsyMachine should explore movement choices", sawMove);
121     }

123     /**
124      * Verifies that OffensiveMachine chases the ball in front or behind, jumps, kicks,
125      * activates full power when close, and backs off when the ball rests on a crossbar.
126      */
127     @Test
128     public void testOffensiveMachineChasesKicksPowersAndBacksOffCrossBar() {
129         OffensiveMachine machine = new OffensiveMachine(Side.RIGHT);
130 
131         StubMatch chaseMatch = new StubMatch()
132                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 270, Field.HEIGHT - 38).withPower(100.0))
133                 .withBall(new StubBall(245, Field.HEIGHT - 12, 0, 0, false));
134         Set<PlayerAction> attackActions = machine.decide(chaseMatch);
135         assertTrue(attackActions.contains(PlayerAction.LEFT));
136         assertTrue(attackActions.contains(PlayerAction.KICK));
137         assertTrue(attackActions.contains(PlayerAction.POWER));
138 
139         StubMatch behindMatch = new StubMatch()
140                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 200, Field.HEIGHT - 38))
141                 .withBall(new StubBall(240, Field.HEIGHT - 55, 0, 0, false));
142         Set<PlayerAction> behindActions = machine.decide(behindMatch);
143         assertTrue("OffensiveMachine should backtrack behind a ball past it",
144                 behindActions.contains(PlayerAction.RIGHT));
145 
146         StubMatch crossbarMatch = new StubMatch()
147                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 35, Field.HEIGHT - 38))
148                 .withBall(new StubBall(15, Field.HEIGHT - 96, 0, 0, false));
149         Set<PlayerAction> backoffActions = machine.decide(crossbarMatch);
150         assertTrue("OffensiveMachine should step back when ball lies on a crossbar",
151                 backoffActions.contains(PlayerAction.RIGHT));
152         assertFalse(backoffActions.contains(PlayerAction.KICK));
153     }

155     /**
156      * Verifies all tactical branches of ExpertMachine: power threat, rival aura, rival under effect,
157      * sudden death, winning, losing, and close-range power activation.
158      */
159     @Test
160     public void testExpertMachineCoversAllTacticalBranches() {
161         ExpertMachine expert = new ExpertMachine(Side.RIGHT);
162 
163         StubMatch losingMatch = new StubMatch()
164                 .withGoals(2, 1)
165                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38))
166                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 270, Field.HEIGHT - 38))
167                 .withBall(new StubBall(100, Field.HEIGHT - 8, 0, 0, false));
168         assertTrue("When losing, Expert should attack toward the ball on the left",
169                 expert.decide(losingMatch).contains(PlayerAction.LEFT));
170 
171         StubMatch winningMatch = new StubMatch()
172                 .withGoals(0, 2)
173                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38))
174                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 200, Field.HEIGHT - 38))
175                 .withBall(new StubBall(80, Field.HEIGHT - 8, 0, 0, false));
176         assertTrue("When winning and ball is far, Expert should retreat defensively",
177                 expert.decide(winningMatch).contains(PlayerAction.RIGHT));
178 
179         StubMatch rivalAuraMatch = new StubMatch()
180                 .withGoals(2, 1)
181                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38).withAura(true))
182                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 200, Field.HEIGHT - 38))
183                 .withBall(new StubBall(80, Field.HEIGHT - 8, 0, 0, false));
184         assertTrue("When rival has aura, Expert should switch to defense",
185                 expert.decide(rivalAuraMatch).contains(PlayerAction.RIGHT));
186 
187         StubMatch powerThreatMatch = new StubMatch()
188                 .withGoals(2, 1)
189                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38))
190                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 200, Field.HEIGHT - 38))
191                 .withBall(new StubBall(120, Field.HEIGHT - 40, 15, 0, true));
192         assertTrue("When a power shot approaches its goal, Expert should defend",
193                 expert.decide(powerThreatMatch).contains(PlayerAction.RIGHT));
194 
195         StubMatch rivalEffectMatch = new StubMatch()
196                 .withGoals(0, 2)
197                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38).withEffect(PlayerEffect.ELECTROCUTED))
198                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 270, Field.HEIGHT - 38))
199                 .withBall(new StubBall(100, Field.HEIGHT - 8, 0, 0, false));
200         assertTrue("When rival is under a power effect, Expert should attack even while winning",
201                 expert.decide(rivalEffectMatch).contains(PlayerAction.LEFT));
202 
203         StubMatch suddenDeathOwnHalf = new StubMatch()
204                 .withSuddenDeath(true)
205                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38))
206                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 240, Field.HEIGHT - 38))
207                 .withBall(new StubBall(260, Field.HEIGHT - 8, 0, 0, false));
208         assertTrue("In sudden death with ball in own half, Expert should move behind ball to defend",
209                 expert.decide(suddenDeathOwnHalf).contains(PlayerAction.RIGHT));
210 
211         StubMatch powerReadyClose = new StubMatch()
212                 .withGoals(1, 1)
213                 .withPlayer(Side.LEFT, new StubPlayer(Side.LEFT, 80, Field.HEIGHT - 38))
214                 .withPlayer(Side.RIGHT, new StubPlayer(Side.RIGHT, 220, Field.HEIGHT - 38).withPower(100.0))
215                 .withBall(new StubBall(195, Field.HEIGHT - 20, 0, 0, false));
216         assertTrue("Expert should activate full power when ball is close and in front",
217                 expert.decide(powerReadyClose).contains(PlayerAction.POWER));
218     }

220     /**
221      * Verifies that a ball resting on top of a goal crossbar rolls back toward the pitch.
222      *
223      * @throws DOPOSoccerException if match configuration is invalid
224      */
225     @Test
226     public void testBallRestingOnCrossBarRollsIntoPitch() throws Exception {
227         MatchConfig config = MatchConfig.builder()
228                 .modality(Modality.PVP)
229                 .leftCountry(Country.SOUTH_KOREA)
230                 .rightCountry(Country.CAMEROON)
231                 .matchType(MatchType.TIME)
232                 .targetValue(1)
233                 .build();
234         Match match = new Match(config);
235         Rectangle leftBar = match.getCrossBar(Side.LEFT);
236         double restX = leftBar.getX() + leftBar.getWidth() / 2.0;
237         double restY = leftBar.getY() - match.getBall().getRadius();
238 
239         java.lang.reflect.Field ballField = Match.class.getDeclaredField("ball");
240         ballField.setAccessible(true);
241         domain.Ball rawBall = (domain.Ball) ballField.get(match);
242         rawBall.reset(restX, restY);
243 
244         match.tick();
245         assertTrue("Ball resting on left crossbar should gain positive speed toward pitch",
246                 match.getBall().getHorizontalSpeed() > 0.0);
247     }
