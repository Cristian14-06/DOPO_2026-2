41       * Verifies that kicking a stationary ball on the ground launches it forward and upward.
42       */
43      @Test
44      public void testKickHitsStationaryBallOnGroundForwardAndUpward() {
45          Player player = new Player("Left", Country.SOUTH_KOREA, Side.LEFT, new AuraAzulPower(), new PlayerStats());
46          double footRestX = player.getFootShape().getX();
47          Ball ball = new Ball(footRestX + 20.0, Field.HEIGHT - 10.0, 10.0);
48  
49          player.kick();
50          for (int i = 0; i < Foot.KICK_SWEEP_TICKS; i++) {
51              player.tick();
52              player.collideWithBall(ball);
53              if (ball.getHorizontalSpeed() > 5.0) {
54                  break;
55              }
56          }
57  
58          assertTrue("Ball should move toward opponent goal (vx > 0)", ball.getHorizontalSpeed() > 5.0);
59          assertTrue("Ball should lift upward (vy < 0)", ball.getVerticalSpeed() < 0.0);
60      }

110     /**
111      * Verifies that jumping directly under the goal crossbar does not trap the player's head inside.
112      */
113     @Test
114     public void testJumpingUnderCrossbarDoesNotTrapHeadInside() {
115         Player player = new Player("Left", Country.SOUTH_KOREA, Side.LEFT, new AuraAzulPower(), new PlayerStats());
116         Goal leftGoal = new Goal(Side.LEFT);
117         Rectangle bar = leftGoal.getCrossBar();
118 
119         double barCenterX = (bar.left() + bar.right()) / 2.0;
120         player.translate(barCenterX - player.getHeadShape().getX(), 0.0);
121         player.jump();
122 
123         int consecutiveInsideTicks = 0;
124         int maxConsecutiveInside = 0;
125 
126         for (int t = 0; t < 120; t++) {
127             player.clearSupport();
128             player.tick();
129             player.collideWithEdges();
130             player.collideWithGoal(leftGoal);
131 
132             double penetration = bar.contactWith(player.getHeadShape())
133                     .map(Contact::getPenetration).orElse(0.0);
134             if (penetration > 0.5) {
135                 consecutiveInsideTicks++;
136                 maxConsecutiveInside = Math.max(maxConsecutiveInside, consecutiveInsideTicks);
137             } else {
138                 consecutiveInsideTicks = 0;
139             }
140         }
141 
142         assertTrue("Head stayed inside crossbar for " + maxConsecutiveInside + " consecutive ticks",
143                 maxConsecutiveInside <= 2);
144     }

208             double footX = player.getFootShape().getX();

232     /*
233      * Drops a ball vertically onto the top of the player's head at a given downward speed
234      * and returns the resulting upward rebound speed magnitude.
235      */
236     private double headerOutwardSpeed(Side side, double incomingDownwardSpeed) {
237         Player player = new Player(new PlayerProfile("Player"), Country.SOUTH_KOREA, side);
238         Circle head = player.getHeadShape();
239         Ball ball = new Ball(head.getX(), head.getY() - head.getRadius() - 8.0, 10.0);
240         ball.setVerticalSpeed(incomingDownwardSpeed);
241         player.collideWithBall(ball);
242         return -ball.getVerticalSpeed();
243     }

317     /*
318      * Simulates a player walking toward their own goal pushing a ground ball and returns true if a goal is detected.
319      */
320     private boolean canPushBallIntoOwnGoal(Side side) {
321         Player player = new Player(new PlayerProfile("Player"), Country.SOUTH_KOREA, side);
322         Goal ownGoal = new Goal(side);
323         double goalEntryX = (side == Side.LEFT) ? 70.0 : Field.WIDTH - 70.0;
324         player.translate((goalEntryX + side.dir() * 25.0) - player.getHeadShape().getX(), 0.0);
325         Ball ball = new Ball(goalEntryX, Field.HEIGHT - 10.0, 10.0);
326 
327         for (int t = 0; t < 60; t++) {
328             if (side == Side.LEFT) {
329                 player.moveLeft();
330             } else {
331                 player.moveRight();
332             }
333             player.tick();
334             ball.tick();
335             ball.collideWithEdges();
336             player.collideWithGoal(ownGoal);
337             player.collideWithBall(ball);
338             if (ownGoal.detectGoal(ball)) {
339                 return true;
340             }
341         }
342         return false;
343     }

345     /**
346      * Verifies exact mirror symmetry between LEFT and RIGHT players when pushing and kicking a ground ball.
347      */
348     @Test
349     public void testLeftAndRightPhysicsSymmetry() {
350         Player left = new Player(new PlayerProfile("Left"), Country.SOUTH_KOREA, Side.LEFT);
351         Player right = new Player(new PlayerProfile("Right"), Country.SOUTH_KOREA, Side.RIGHT);
352 
353         Ball leftBall = new Ball(left.getFootShape().getX() + 18.0, Field.HEIGHT - 10.0, 10.0);
354         Ball rightBall = new Ball(right.getFootShape().getX() - 18.0, Field.HEIGHT - 10.0, 10.0);
355 
356         left.kick();
357         right.kick();
358         double leftMinVerticalSpeed = 0.0;
359         double rightMinVerticalSpeed = 0.0;
360         for (int t = 0; t < 10; t++) {
361             left.tick();
362             right.tick();
363             leftBall.tick();
364             rightBall.tick();
365             leftBall.collideWithEdges();
366             rightBall.collideWithEdges();
367             left.collideWithBall(leftBall);
368             right.collideWithBall(rightBall);
369             leftMinVerticalSpeed = Math.min(leftMinVerticalSpeed, leftBall.getVerticalSpeed());
370             rightMinVerticalSpeed = Math.min(rightMinVerticalSpeed, rightBall.getVerticalSpeed());
371         }
372 
373         assertTrue("Left player kick should launch ball right", leftBall.getHorizontalSpeed() > 5.0);
374         assertTrue("Right player kick should launch ball left", rightBall.getHorizontalSpeed() < -5.0);
375         assertTrue("Left player kick should lift ball upward", leftMinVerticalSpeed < 0.0);
376         assertTrue("Right player kick should lift ball upward", rightMinVerticalSpeed < 0.0);
377     }

379     /**
380      * Verifies that when two adjacent players simultaneously kick a ball between them,
381      * the ball pops upward and never remains overlapping either player's body.
382      *
383      * @throws DOPOSoccerException if match configuration is invalid
384      */
385     @Test
386     public void testSimultaneousKickPinchPopsBallUpwardAndLeavesNoOverlap() throws DOPOSoccerException {
387         MatchConfig config = MatchConfig.builder()
388                 .modality(Modality.PVP)
389                 .leftCountry(Country.SOUTH_KOREA)
390                 .rightCountry(Country.CAMEROON)
391                 .matchType(MatchType.TIME)
392                 .targetValue(1)
393                 .build();
394         Match match = new Match(config);
395 
396         boolean poppedUp = false;
397         for (int t = 0; t < 120; t++) {
398             match.setIntention(Side.LEFT, EnumSet.of(PlayerAction.RIGHT, PlayerAction.KICK));
399             match.setIntention(Side.RIGHT, EnumSet.of(PlayerAction.LEFT, PlayerAction.KICK));
400             match.tick();
401 
402             BallView b = match.getBall();
403             if (b.getVerticalSpeed() < -5.0 && Math.abs(b.getHorizontalSpeed()) < 0.01) {
404                 poppedUp = true;
405                 Circle ballCircle = new Circle(b.getX(), b.getY(), b.getRadius());
406                 PlayerView p1 = match.getPlayer(Side.LEFT);
407                 PlayerView p2 = match.getPlayer(Side.RIGHT);
408                 assertFalse("Ball should not remain inside left player after pinch escape",
409                         ballCircle.detectCollisionWithCircle(p1.getHeadShape())
410                                 || ballCircle.detectCollisionWithRectangle(p1.getBodyShape())
411                                 || ballCircle.detectCollisionWithCircle(p1.getFootShape()));
412                 assertFalse("Ball should not remain inside right player after pinch escape",
413                         ballCircle.detectCollisionWithCircle(p2.getHeadShape())
414                                 || ballCircle.detectCollisionWithRectangle(p2.getBodyShape())
415                                 || ballCircle.detectCollisionWithCircle(p2.getFootShape()));
416                 break;
417             }
418         }
419         assertTrue("Ball pinched between both kicking players should pop upward", poppedUp);
420     }
