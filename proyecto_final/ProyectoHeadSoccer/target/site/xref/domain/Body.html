5   /**
6    * Represents the rectangular torso/body of a player.
7    * 
8    * @author Juan Espitia
9    * @author Cristian Salamanca
10   */
11  public class Body implements Serializable {
12  	private static final long serialVersionUID = 1L;
13  	private Rectangle rectangle;
14  
15  	/**
16  	 * Constructs the body aligned to the player's horizontal coordinate and side.
17  	 * 
18  	 * @param playerX Player horizontal center position
19  	 */
20  	public Body(double playerX) {
21  		double width = 18;
22  		double height = 16;
23  		double bodyX = playerX - (width / 2.0);
24  		double bodyY = Field.HEIGHT - 26;
25  		this.rectangle = new Rectangle(bodyX, bodyY, width, height);
26  	}
27  
28  	/**
29  	 * Gets the rectangular geometry of the body for domain physics.
30  	 *
31  	 * @return Rectangle geometry
32  	 */
33  	Rectangle getRectangle() {
34  		return rectangle;
35  	}
36  
37  	/**
38  	 * Sets the rectangular geometry of the body.
39  	 * 
40  	 * @param rectangle New rectangle geometry
41  	 */
42  	public void setRectangle(Rectangle rectangle) {
43  		this.rectangle = rectangle;
44  	}
45  }
