1   package domain;
2   
3   import java.util.EnumSet;
4   import java.util.Random;
5   import java.util.Set;
6   
7   /**
8    * Machine controller implementing the Torpe (clumsy) profile with randomized,
9    * erratic movements and delayed reactions.
10   *
11   * @author Juan Espitia
12   * @author Cristian Salamanca
13   */
14  public class ClumsyMachine extends MachineController {
15      private static final long serialVersionUID = 1L;
16      private static final int MIN_ACTION_TICKS = 12;
17      private static final int EXTRA_RANDOM_TICKS = 18;
18      private static final double POWER_CHANCE = 0.10;
19      private static final double FULL_POWER = 100.0;
20      private static final int CHOICES = 6;
21  
22      private final Random random;
23      private int actionTimer = 0;
24      private final Set<PlayerAction> bufferedActions = EnumSet.noneOf(PlayerAction.class);
25  
26      /**
27       * Constructs a clumsy machine controller with a default random generator.
28       *
29       * @param side Pitch side controlled by this AI
30       */
31      public ClumsyMachine(Side side) {
32          this(side, new Random());
33      }
34  
35      /**
36       * Constructs a clumsy machine controller with an injectable random generator for testing.
37       *
38       * @param side Pitch side controlled by this AI
39       * @param random Random generator instance
40       */
41      public ClumsyMachine(Side side, Random random) {
42          super(side);
43          this.random = (random != null) ? random : new Random();
44      }
45  
46      /**
47       * Picks a random action and keeps it for 12 to 29 ticks, so the machine looks clumsy instead of shaking.
48       * The possible choices are: move left, move right, jump, kick, move roughly toward the ball
49       * (sometimes the wrong way) with an occasional kick, or stay idle for a moment.
50       * When the power bar is full it activates the power with a 10% chance per decision.
51       *
52       * @param match Read-only match view
53       * @param me Read-only view of the controlled player
54       * @param ball Read-only ball view
55       * @param rival Read-only view of the opponent
56       * @return Actions to perform during this tick
57       */
58      @Override
59      protected Set<PlayerAction> chooseActions(MatchView match, PlayerView me, BallView ball, PlayerView rival) {
60          if (actionTimer > 0) {
61              actionTimer--;
62              return EnumSet.copyOf(bufferedActions);
63          }
64  
65          bufferedActions.clear();
66          actionTimer = MIN_ACTION_TICKS + random.nextInt(EXTRA_RANDOM_TICKS);
67  
68          int choice = random.nextInt(CHOICES);
69          switch (choice) {
70              case 0:
71                  bufferedActions.add(PlayerAction.LEFT);
72                  break;
73              case 1:
74                  bufferedActions.add(PlayerAction.RIGHT);
75                  break;
76              case 2:
77                  bufferedActions.add(PlayerAction.JUMP);
78                  break;
79              case 3:
80                  bufferedActions.add(PlayerAction.KICK);
81                  break;
82              case 4:
83                  if (ball.getX() > me.getHeadShape().getX()) {
84                      bufferedActions.add(random.nextBoolean() ? PlayerAction.RIGHT : PlayerAction.LEFT);
85                  } else {
86                      bufferedActions.add(random.nextBoolean() ? PlayerAction.LEFT : PlayerAction.RIGHT);
87                  }
88                  if (random.nextBoolean()) {
89                      bufferedActions.add(PlayerAction.KICK);
90                  }
91                  break;
92              case 5:
93              default:
94                  break;
95          }
96  
97          if (me.getPowerProgress() >= FULL_POWER && random.nextDouble() < POWER_CHANCE) {
98              bufferedActions.add(PlayerAction.POWER);
99          }
100 
101         return EnumSet.copyOf(bufferedActions);
102     }
103 }
