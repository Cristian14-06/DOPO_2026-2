3   /**
4    * Immutable collision contact describing the outward normal from the obstacle
5    * toward the ball and the overlap penetration depth.
6    *
7    * @author Juan Espitia
8    * @author Cristian Salamanca
9    */
10  public class Contact {
11      private final Vector2 normal;
12      private final double penetration;
13  
14      /**
15       * Constructs a collision contact with a normal vector and penetration depth.
16       *
17       * @param normal Outward unit normal vector from the obstacle toward the ball
18       * @param penetration Overlap distance in pixels
19       */
20      public Contact(Vector2 normal, double penetration) {
21          this.normal = normal;
22          this.penetration = penetration;
23      }
24  
25      /**
26       * Returns the outward normal vector of this contact.
27       *
28       * @return Normal vector
29       */
30      public Vector2 getNormal() {
31          return normal;
32      }
33  
34      /**
35       * Returns the overlap penetration depth in pixels.
36       *
37       * @return Penetration depth
38       */
39      public double getPenetration() {
40          return penetration;
41      }
42  }
