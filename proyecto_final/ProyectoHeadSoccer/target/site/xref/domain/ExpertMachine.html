1   package domain;
2   
3   import java.util.EnumSet;
4   import java.util.Set;
5   
6   /**
7    * Machine controller implementing the Experta (expert) profile adapting tactics to match state.
8    *
9    * @author Juan Espitia
10   * @author Cristian Salamanca
11   */
12  public class ExpertMachine extends MachineController {
13      private static final long serialVersionUID = 1L;
14      private static final int LOW_TIME_SECONDS = 15;
15      private static final double POWER_BALL_DISTANCE = 45.0;
16      private static final double FULL_POWER = 100.0;
17      private static final double OWN_HALF_FACTOR = 0.45;
18  
19      private final OffensiveMachine offensive;
20      private final DefensiveMachine defensive;
21  
22      /**
23       * Constructs an expert machine controller for the given pitch side.
24       *
25       * @param side Pitch side controlled by this AI
26       */
27      public ExpertMachine(Side side) {
28          super(side);
29          this.defensive = new DefensiveMachine(side);
30          this.offensive = new OffensiveMachine(side);
31      }
32  
33      /**
34       * Chooses between the defensive and the offensive behavior, in this order:
35       * <ul>
36       *   <li>Defends if a power shot is coming toward its goal or the rival has the aura.</li>
37       *   <li>Attacks if the rival is stunned or affected by any power status effect.</li>
38       *   <li>In sudden death, defends while the ball is in its own half and attacks otherwise.</li>
39       *   <li>Defends while winning, unless the score is tied with little time left.</li>
40       *   <li>Attacks in any other case.</li>
41       * </ul>
42       * It also activates the power when the bar is full and the ball is close and in front.
43       *
44       * @param match Read-only match view
45       * @param me Read-only view of the controlled player
46       * @param ball Read-only ball view
47       * @param rival Read-only view of the opponent
48       * @return Actions to perform during this tick
49       */
50      @Override
51      protected Set<PlayerAction> chooseActions(MatchView match, PlayerView me, BallView ball, PlayerView rival) {
52          int myGoals = match.getGoals(side);
53          int rivalGoals = match.getGoals(side.opposite());
54          int remaining = match.getRemainingSeconds();
55          boolean lowTimeTie = (myGoals == rivalGoals) && remaining >= 0 && remaining <= LOW_TIME_SECONDS;
56  
57          MachineController chosen;
58          boolean ballPowerThreat = ball.hasPowerShot() && (ball.getHorizontalSpeed() * side.dir() < 0);
59          boolean rivalUnderEffect = rival != null && (rival.isStunned()
60                  || (rival.getActiveEffect() != null && rival.getActiveEffect() != PlayerEffect.NONE));
61  
62          if (ballPowerThreat || (rival != null && rival.hasAura())) {
63              chosen = defensive;
64          } else if (rivalUnderEffect) {
65              chosen = offensive;
66          } else if (match.isSuddenDeath()) {
67              double distToGoal = Math.abs(ball.getX() - ownGoalX(match));
68              chosen = (distToGoal < match.getFieldWidth() * OWN_HALF_FACTOR) ? defensive : offensive;
69          } else if (myGoals > rivalGoals && !lowTimeTie) {
70              chosen = defensive;
71          } else {
72              chosen = offensive;
73          }
74  
75          Set<PlayerAction> actions = EnumSet.noneOf(PlayerAction.class);
76          actions.addAll(chosen.chooseActions(match, me, ball, rival));
77  
78          if (me.getPowerProgress() >= FULL_POWER) {
79              double myX = me.getHeadShape().getX();
80              double distBall = Math.hypot(myX - ball.getX(), me.getHeadShape().getY() - ball.getY());
81              if (distBall <= POWER_BALL_DISTANCE && (ball.getX() - myX) * side.dir() >= 0) {
82                  actions.add(PlayerAction.POWER);
83              }
84          }
85          return actions;
86      }
87  }
