1   package domain;
2   
3   import java.util.List;
4   import java.util.Optional;
5   
6   /**
7    * Read-only view of the match state for presentation and AI controllers.
8    * 
9    * @author Juan Espitia
10   * @author Cristian Salamanca
11   */
12  public interface MatchView {
13  
14      /**
15       * Returns the current lifecycle status of the match.
16       *
17       * @return Active GameStatus
18       */
19      GameStatus getStatus();
20  
21      /**
22       * Returns the remaining match time in seconds, or -1 for goal-based matches.
23       *
24       * @return Remaining seconds or -1
25       */
26      int getRemainingSeconds();
27  
28      /**
29       * Indicates whether the match is currently in sudden-death golden goal mode.
30       *
31       * @return True if sudden death is active
32       */
33      boolean isSuddenDeath();
34  
35      /**
36       * Returns the number of goals scored by the specified side.
37       *
38       * @param side Target side
39       * @return Goal count for the side
40       */
41      int getGoals(Side side);
42  
43      /**
44       * Returns a read-only view of the player on the specified side.
45       *
46       * @param side Target side
47       * @return PlayerView for the side
48       */
49      PlayerView getPlayer(Side side);
50  
51      /**
52       * Returns a read-only view of the ball in play.
53       *
54       * @return BallView instance
55       */
56      BallView getBall();
57  
58      /**
59       * Returns the winning side if the match has concluded with a winner.
60       *
61       * @return Optional containing the winning side, or empty if undecided or tied
62       */
63      Optional<Side> getWinner();
64  
65      /**
66       * Returns the logical width of the playing field.
67       *
68       * @return Field width in logical units
69       */
70      int getFieldWidth();
71  
72      /**
73       * Returns the logical height of the playing field.
74       *
75       * @return Field height in logical units
76       */
77      int getFieldHeight();
78  
79      /**
80       * Returns a defensive copy of the goal scoring area for the given side.
81       *
82       * @param side Goal side
83       * @return Goal area rectangle
84       */
85      Rectangle getGoalArea(Side side);
86  
87      /**
88       * Returns a defensive copy of the goal crossbar hitbox for the given side.
89       *
90       * @param side Goal side
91       * @return Crossbar rectangle
92       */
93      Rectangle getCrossBar(Side side);
94  
95      /**
96       * Returns the game modality configured for this match.
97       *
98       * @return Active Modality
99       */
100     Modality getModality();
101 
102     /**
103      * Returns defensive copies of any active illusion decoy balls on the field.
104      *
105      * @return List of decoy ball circles
106      */
107     List<Circle> getFakeBalls();
108 }
