39  	/**
40  	 * Constructs a match player backed by a persistent player profile.
41  	 *
42  	 * @param profile Persistent player profile
43  	 * @param country Country representation
44  	 * @param side Pitch side assigned
45  	 * @param power Special power ability
46  	 */
47  	public Player(PlayerProfile profile, Country country, Side side, Power power) {
48  		this.profile = (profile != null) ? profile : new PlayerProfile("Player");
49  		this.country = country;
50  		this.side = side;
51  		this.power = power;
52  		this.playerBody = new PlayerBody(side);
53  		this.playerBody.getFoot().setPowerKicking((int) this.profile.getStats().kickBallForce());
54  		this.life = new Life(this.profile.getStats().getMaxLife());
55  		this.effect = new EffectState();
56  	}

92  	 *

96  		return life.isStunned() || !effect.canMove()

100 	/**
101 	 * Moves player horizontally to the left according to speed stat and status effects.
102 	 */
103 	public void moveLeft() {
104 		if (isDisabled()) return;
105 		if (effect.isConfused()) {
106 			moveRightInternal();
107 			return;
108 		}
109 		moveLeftInternal();
110 	}
111 
112 	/**
113 	 * Moves player horizontally to the right according to speed stat and status effects.
114 	 */
115 	public void moveRight() {
116 		if (isDisabled()) return;
117 		if (effect.isConfused()) {
118 			moveLeftInternal();
119 			return;
120 		}
121 		moveRightInternal();
122 	}

124 	private void moveLeftInternal() {
125 		double spd = profile.getStats().speedPx() * effect.speedFactor();
126 		playerBody.moveHorizontally(-spd);
127 		playerBody.collideWithEdges();
128 	}
129 
130 	private void moveRightInternal() {
131 		double spd = profile.getStats().speedPx() * effect.speedFactor();
132 		playerBody.moveHorizontally(spd);
133 		playerBody.collideWithEdges();
134 	}

226 	/**
227 	 * Applies a set of intended input actions onto the player.
228 	 * 
229 	 * @param actions Set of requested player actions
230 	 * @param ball Ball reference for shooting actions
231 	 * @param rival Opponent player reference for power effects
232 	 */
233 	public void applyInput(Set<PlayerAction> actions, Ball ball, Player rival) {
234 		if (actions == null) {
235 			wasKicking = false;
236 			wasPowerPressed = false;
237 			return;
238 		}
239 
240 		boolean kickingNow = actions.contains(PlayerAction.KICK);
241 		boolean powerNow = actions.contains(PlayerAction.POWER);
242 
243 		for (PlayerAction action : actions) {
244 			switch (action) {
245 				case LEFT:
246 					moveLeft();
247 					break;
248 				case RIGHT:
249 					moveRight();
250 					break;
251 				case JUMP:
252 					jump();
253 					break;
254 				case KICK:
255 					if (!wasKicking) {
256 						kick();
257 					}
258 					break;
259 				case POWER:
260 					if (!wasPowerPressed) {
261 						activatePower();
262 					}
263 					break;
264 			}
265 		}
266 
267 		wasKicking = kickingNow;
268 		wasPowerPressed = powerNow;
269 	}

287 	/**
288 	 * Resolves collision, stacking, and push mechanics between this player and an opponent.
289 	 * 
290 	 * @param player Opponent player to collide with
291 	 */
292 	public void collideWithPlayer(Player player) {
293 		if (player == null) return;
294 		playerBody.collideWithPlayer(player.playerBody, player.profile.getStats().speedPx());
295 
296 		if (!kickDealtDamage && !isDisabled() && playerBody.isKickHitting(player.playerBody)) {
297 			player.takeDamage(profile.getStats().getKickDamage());
298 			kickDealtDamage = true;
299 		}
300 	}

318 	/**
319 	 * Resolves ball collisions against this player's body, head, and foot.
320 	 *
321 	 * @param ball Ball in play
322 	 */
323 	public boolean collideWithBall(Ball ball) {
324 		boolean hadArua = hasAura();
325 		boolean collide = playerBody.collideWithBall(ball, power);
326 		if(hadArua && !hasAura()){
327 			ball.setPowerShotCountry(country);
328 		}
329 		return collide;
330 	}

342 	 * Applies damage and status effect from an opponent power hit.
343 	 *
344 	 * @param hit Power hit payload to apply
345 	 */
346 	public void receiveHit(PowerHit hit) {
347 		if (hit == null || hasAura()) return;
348 		if (hit.getDamage() > 0) {
349 			takeDamage(hit.getDamage());
350 		}
351 		if (hit.getEffect() != null && hit.getEffect().getType() != PlayerEffect.NONE) {
352 			effect.apply(hit.getEffect(), hit.getDurationSeconds(), playerBody);
353 		}
354 	}

387 	 * Evaluates whether this player's active power shot hits the opponent and returns the hit payload.
388 	 *
389 	 * @param rival Opponent player
390 	 * @param ball Ball in play
391 	 * @return PowerHit payload if hit, or null otherwise
392 	 */
393 	public PowerHit powerHitOn(Player rival, Ball ball) {
394 		if (power == null || rival == null || !power.isPowerShotActive() || !rival.isHitByBall(ball)) {
395 			return null;
396 		}
397 		return power.onHitRival(getKickMultiplier());
398 	}
