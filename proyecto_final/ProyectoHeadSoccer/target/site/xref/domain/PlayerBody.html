7   /**
8    * Encapsulates a player's physical limbs (head, torso, foot), anchor position,
9    * vertical kinematics, and collision resolution against field boundaries, goals,
10   * the ball, and opponents.
11   *
12   * @author Juan Espitia
13   * @author Cristian Salamanca
14   */
15  public class PlayerBody implements Serializable {
16      private static final long serialVersionUID = 1L;
17      private final Side side;
18      private final Head head;
19      private final Body body;
20      private final Foot foot;
21      private Vector2 anchor;
22      private double verticalSpeed;
23      private boolean onTop;
24      private double lastHorizontalMove;
25      private static final double DRIBBLE_SPEED_FACTOR = 1.5;
26      private static final double KICK_LIFT_HIGHEST = -1.45;
27      private static final double KICK_LIFT_LOWEST = -0.45;
28      private final Random kickRandom;
29  
30      /*
31       * Synchronizes head, torso, and foot collision shapes with the current anchor and kick offsets.
32       */
33      private void syncShapes() {
34          head.getCircle().setX(anchor.getX());
35          head.getCircle().setY(anchor.getY() - 38);
36          
37          body.getRectangle().setX(anchor.getX() - 9);
38          body.getRectangle().setY(anchor.getY() - 26);
39  
40          foot.getCircle().setX(anchor.getX() + side.dir()*6 + foot.getKickOffsetX(side));
41          foot.getCircle().setY(anchor.getY() - 8 + foot.getKickOffsetY());
42          
43      }

45      /**
46       * Constructs a player body at the starting position for the given side.
47       *
48       * @param side Field side defended by this player
49       */
50      public PlayerBody(Side side){
51          this(side, new Random());
52      }

54      /**
55       * Constructs a player body with an injectable random generator for the kick angle variation,
56       * so tests can reproduce the exact same kicks with a fixed seed.
57       *
58       * @param side Field side defended by this player
59       * @param kickRandom Random generator used to vary the kick lift; a new one is created if null
60       */
61      public PlayerBody(Side side, Random kickRandom){
62          this.side = side;
63          this.kickRandom = (kickRandom != null) ? kickRandom : new Random();
64          anchor = new Vector2(side.startX(), Field.HEIGHT);
65          head = new Head(anchor.getX());
66          body = new Body(anchor.getX());
67          foot = new Foot(anchor.getX(), side);
68          verticalSpeed = 0;
69          onTop = false;
70          syncShapes();
71      }
72  
73      /**
74       * Resets the body to its starting anchor position and clears velocity and kick state.
75       */
76      public void resetPosition() {
77          anchor = new Vector2(side.startX(), Field.HEIGHT);
78          verticalSpeed = 0;
79          onTop = false;
80          foot.stopKicking(side);
81          syncShapes();
82          this.lastHorizontalMove = 0;
83      }

85      /**
86       * Translates the entire body by the given delta.
87       *
88       * @param dx Horizontal displacement in pixels
89       * @param dy Vertical displacement in pixels
90       */
91      public void translate(double dx, double dy) {
92          anchor = anchor.add(new Vector2(dx, dy));
93          syncShapes();
94      }

96      /**
97       * Translates the player horizontally and records the movement delta for ground dribbling.
98       *
99       * @param dx Horizontal displacement in pixels
100      */
101     public void moveHorizontally(double dx) {
102         this.lastHorizontalMove = dx;
103         translate(dx, 0);
104     }

106     /**
107      * Applies an upward jump impulse if the player is grounded or standing on a surface.
108      *
109      * @param jumpImpulse Upward impulse magnitude in pixels per tick
110      */
111     public void jump(double jumpImpulse) {
112         if (anchor.getY() >= Field.HEIGHT - 2.0 || onTop) {
113             verticalSpeed = -jumpImpulse;
114             onTop = false;
115         }
116     }

118     /**
119      * Advances foot kick sweep and vertical kinematics (gravity, ceiling, and ground clamp).
120      */
121     public void tick() {
122         foot.tick(side);
123         translate(0, verticalSpeed);
124 
125         double topY = head.getCircle().getY() - head.getCircle().getRadius();
126         if (topY < 0) {
127             translate(0, -topY);
128             if (verticalSpeed < 0) {
129                 verticalSpeed = 0;
130             }
131         }
132 
133         if (anchor.getY() < Field.HEIGHT && !onTop) {
134             verticalSpeed += Physics.GRAVITY;
135         } else {
136             if (anchor.getY() > Field.HEIGHT) {
137                 translate(0, Field.HEIGHT - anchor.getY());
138             }
139             verticalSpeed = 0;
140         }
141     }

143     /**
144      * Resolves solid collision between head, body, and foot against a goal crossbar.
145      *
146      * @param goal Goal to test against
147      */
148     public void collideWithGoal(Goal goal) {
149         if (goal == null || goal.getCrossBar() == null) return;
150         Rectangle bar = goal.getCrossBar();
151         resolveCircleAgainstBar(head.getCircle(), bar);
152         resolveCircleAgainstBar(foot.getCircle(), bar);
153         resolveBodyAgainstBar(body.getRectangle(), bar);
154     }
155 
156     /*
157      * Pushes a circular limb out of the goal crossbar and updates vertical support state.
158      */
159     private void resolveCircleAgainstBar(Circle circle, Rectangle bar) {
160         Optional<Contact> contactOpt = bar.contactWith(circle);
161         if (!contactOpt.isPresent()) return;
162 
163         Contact c = contactOpt.get();
164         Vector2 n = c.getNormal();
165         translate(n.getX() * c.getPenetration(), n.getY() * c.getPenetration());
166 
167         if (n.getY() < -0.5 && verticalSpeed >= 0) {
168             verticalSpeed = 0;
169             onTop = true;
170         } else if (n.getY() > 0.5) {
171             verticalSpeed = Math.max(verticalSpeed, 0);
172         }
173     }
174 
175     /*
176      * Resolves AABB overlap between the player's torso rectangle and the goal crossbar.
177      */
178     private void resolveBodyAgainstBar(Rectangle br, Rectangle bar) {
179         double overlapX = Math.min(br.right(), bar.right()) - Math.max(br.left(), bar.left());
180         double overlapY = Math.min(br.bottom(), bar.bottom()) - Math.max(br.top(), bar.top());
181         if (overlapX <= 0 || overlapY <= 0) return;
182 
183         double bodyCenterX = (br.left() + br.right()) / 2.0;
184         double barCenterX = (bar.left() + bar.right()) / 2.0;
185         double bodyCenterY = (br.top() + br.bottom()) / 2.0;
186         double barCenterY = (bar.top() + bar.bottom()) / 2.0;
187 
188         if (overlapX < overlapY) {
189             translate(bodyCenterX < barCenterX ? -overlapX : overlapX, 0);
190         } else if (bodyCenterY < barCenterY) {
191             translate(0, -overlapY);
192             if (verticalSpeed >= 0) {
193                 verticalSpeed = 0;
194                 onTop = true;
195             }
196         } else {
197             translate(0, overlapY);
198             verticalSpeed = Math.max(verticalSpeed, 0);
199         }
200     }

202     /**
203      * Resolves ball collisions against this body, prioritizing the foot while kicking.
204      *
205      * @param ball Ball in play
206      * @param power Player power to trigger if aura is active on kick contact
207      */
208     public boolean collideWithBall(Ball ball, Power power){
209         boolean collide;
210         boolean hitHead = false;
211         boolean hitBody = false;
212         if (ball == null) return false;
213 
214         if (ball.isPowerShotCharging()) {
215             if (power == null || !power.isChargingShot()) {
216                 pushOutOfChargingBall(ball.getCircle());
217             }
218             this.lastHorizontalMove = 0.0;
219             return false;
220         }
221 
222         if (foot.isKicking()) {
223             collide = collideFootWithBall(ball, power);
224             if (!ball.isPowerShotCharging()) {
225                 hitHead = ball.bounceOff(head.getCircle(), Physics.HEAD_RESTITUTION, Physics.HEAD_IMPULSE);
226                 hitBody = ball.bounceOff(body.getRectangle(), Physics.BODY_RESTITUTION, 0.0);
227             }
228         } else {
229             hitHead = ball.bounceOff(head.getCircle(), Physics.HEAD_RESTITUTION, Physics.HEAD_IMPULSE);
230             hitBody = ball.bounceOff(body.getRectangle(), Physics.BODY_RESTITUTION, 0.0);
231             collide = collideFootWithBall(ball, power);
232         }
233         if (hitHead || hitBody) {
234             triggerPowerShot(ball, power);
235         }
236         this.lastHorizontalMove = 0.0;
237         return collide;
238     }

240     /*
241      * Pushes all limbs horizontally out of a suspended charging ball so opponents cannot walk through it.
242      */
243     private void pushOutOfChargingBall(Circle ballCircle) {
244         if (ballCircle == null) return;
245         pushCircleOutHorizontally(this.head.getCircle(), ballCircle);
246         pushCircleOutHorizontally(this.foot.getCircle(), ballCircle);
247         pushBodyOutHorizontally(this.body.getRectangle(), ballCircle);
248         collideWithEdges();
249     }

251     /*
252      * Separates a circular limb horizontally from a stationary charging ball using exact circle clearance.
253      */
254     private void pushCircleOutHorizontally(Circle circle, Circle ballCircle) {
255         double dy = circle.getY() - ballCircle.getY();
256         double sumRad = circle.getRadius() + ballCircle.getRadius();
257         if (Math.abs(dy) >= sumRad) return;
258         double minDx = Math.sqrt(sumRad * sumRad - dy * dy);
259         double dx = circle.getX() - ballCircle.getX();
260         if (Math.abs(dx) < minDx) {
261             double pushDir = (dx != 0.0) ? Math.signum(dx) : -side.dir();
262             translate(pushDir * (minDx - Math.abs(dx) + 0.05), 0.0);
263         }
264     }
265 
266     /*
267      * Separates the rectangular torso horizontally from a stationary charging ball.
268      */
269     private void pushBodyOutHorizontally(Rectangle br, Circle ballCircle) {
270         if (!br.detectCollisionWithCircle(ballCircle)) return;
271         double closestY = Math.max(br.top(), Math.min(ballCircle.getY(), br.bottom()));
272         double dy = ballCircle.getY() - closestY;
273         double r = ballCircle.getRadius();
274         if (Math.abs(dy) >= r) return;
275         double clearanceX = Math.sqrt(r * r - dy * dy);
276         if (anchor.getX() >= ballCircle.getX()) {
277             double targetLeft = ballCircle.getX() + clearanceX + 0.05;
278             if (br.left() < targetLeft) {
279                 translate(targetLeft - br.left(), 0.0);
280             }
281         } else {
282             double targetRight = ballCircle.getX() - clearanceX - 0.05;
283             if (br.right() > targetRight) {
284                 translate(targetRight - br.right(), 0.0);
285             }
286         }
287     }

289     /*
290      * Triggers the player's power shot holding the ball in front of the head toward the opponent's goal.
291      */
292     private void triggerPowerShot(Ball ball, Power power) {
293         if (power == null || !power.isAuraActive()) return;
294         Circle hc = head.getCircle();
295         double holdX = hc.getX() + side.dir() * (hc.getRadius() + ball.getRadius() + 4.0);
296         double holdY = hc.getY();
297         power.executePowerShot(ball, side, holdX, holdY);
298     }

300     /*
301      * Resolves foot-to-ball contact, launching a kick or power shot when kicking in front of the ball.
302      */
303     private boolean collideFootWithBall(Ball ball, Power power) {
304         boolean hitFoot = false;
305         Circle footCircle = this.foot.getCircle();
306         Optional<Contact> contact = footCircle.contactWith(ball.getCircle());
307         if (contact.isEmpty()) return false;
308         Contact c = contact.get();
309         if (foot.isKicking()) {
310             boolean inFront = (ball.getX() - footCircle.getX()) * side.dir() >= -ball.getRadius();
311             if (inFront) {
312                 ball.getCircle().translate(side.dir() * c.getPenetration(), 0);
313                 double lift = KICK_LIFT_HIGHEST + kickRandom.nextDouble() * (KICK_LIFT_LOWEST - KICK_LIFT_HIGHEST);
314                 Vector2 vel = new Vector2(side.dir(), lift).normalized().scale(foot.getPowerKicking());
315                 ball.setHorizontalSpeed(vel.getX());
316                 ball.setVerticalSpeed(vel.getY());
317                 triggerPowerShot(ball, power);
318                 return true;
319             }
320         }
321         hitFoot = ball.bounceOff(footCircle, Physics.FOOT_RESTITUTION, 0.0);
322         if (hitFoot && Math.abs(this.lastHorizontalMove) > 0.01 && ball.getY() + ball.getRadius() >= Field.HEIGHT - 2.0 && (ball.getX() - footCircle.getX()) * lastHorizontalMove > 0) {
323             double dir = Math.signum(lastHorizontalMove);
324             double dribbleSpeed = Math.abs(lastHorizontalMove) * DRIBBLE_SPEED_FACTOR;
325             if (ball.getHorizontalSpeed() * dir < dribbleSpeed) {
326                 ball.setHorizontalSpeed(dir * dribbleSpeed);
327             }
328         }
329         if (hitFoot) {
330             triggerPowerShot(ball, power);
331         }
332 
333         return false;
334     }

337      * Checks whether the ball overlaps any limb of this body.
338      *
339      * @param ball Ball in play
340      * @return True if any limb overlaps the ball
341      */
342     public boolean isHitByBall(Ball ball) {
343         if (ball == null || ball.getCircle() == null) return false;
344         Circle ballCircle = ball.getCircle();
345         return ballCircle.detectCollisionWithCircle(head.getCircle())
346                 || ballCircle.detectCollisionWithRectangle(body.getRectangle())
347                 || ballCircle.detectCollisionWithCircle(foot.getCircle());
348     }

416     /**
417      * Restricts the player horizontally within the pitch boundaries.
418      */
419     public void collideWithEdges() {
420         double r = head.getCircle().getRadius();
421         double hx = head.getCircle().getX();
422         if (hx - r < 0) {
423             translate(-(hx - r), 0);
424         }
425         if (hx + r > Field.WIDTH) {
426             translate(Field.WIDTH - (hx + r), 0);
427         }
428     }

430     /**
431      * Resolves physical separation and head-standing support against an opponent body.
432      *
433      * @param other Opponent body
434      * @param pushSpeed Horizontal separation speed
435      */
436     public void collideWithPlayer(PlayerBody other, double pushSpeed) {
437         if (other == null) return;
438         boolean overlaps = head.getCircle().detectCollisionWithCircle(other.head.getCircle())
439                 || head.getCircle().detectCollisionWithRectangle(other.body.getRectangle())
440                 || body.getRectangle().detectCollisionWithCircle(other.head.getCircle())
441                 || body.getRectangle().detectCollisionWithRectangle(other.body.getRectangle())
442                 || foot.getCircle().detectCollisionWithCircle(other.head.getCircle())
443                 || foot.getCircle().detectCollisionWithCircle(other.foot.getCircle())
444                 || foot.getCircle().detectCollisionWithRectangle(other.body.getRectangle())
445                 || head.getCircle().detectCollisionWithCircle(other.foot.getCircle());
446 
447         if (!overlaps) return;
448 
449         if (foot.getCircle().detectCollisionWithCircle(other.head.getCircle())
450                 && foot.getCircle().getY() < other.head.getCircle().getY()) {
451             verticalSpeed = 0;
452             onTop = true;
453         }
454 
455         double push = (anchor.getX() < other.anchor.getX()) ? -pushSpeed : pushSpeed;
456         translate(push, 0);
457         collideWithEdges();
458     }
