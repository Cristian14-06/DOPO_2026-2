3   /**
4    * Effect strategy applied by AuraAzulPower: pushes the opponent toward their own goal each tick.
5    *
6    * @author Juan Espitia
7    * @author Cristian Salamanca
8    */
9   public class PushedEffect implements Effect{
10      /**
11       * Returns the PUSHED effect type.
12       *
13       * @return PlayerEffect.PUSHED
14       */
15      @Override 
16      public PlayerEffect getType(){
17          return PlayerEffect.PUSHED;
18      }
19  
20      /**
21       * Translates the player toward their own goal and clamps to pitch boundaries.
22       *
23       * @param body Player body affected
24       * @param side Pitch side of the affected player
25       */
26      @Override
27      public void onTick(PlayerBody body, Side side){
28          if(body == null || side == null) return;
29          body.translate(-4.0* side.dir(), 0.0);
30          body.collideWithEdges();
31      }
32  }
