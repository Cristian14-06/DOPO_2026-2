3   /**
4    * Enumeration representing pitch sides (LEFT or RIGHT) assigned to players and goals.
5    * 
6    * @author Juan Espitia
7    * @author Cristian Salamanca
8    */
9    public enum Side {
10       LEFT(1, 100.0),
11       RIGHT(-1, Field.WIDTH - 100.0);
12       private final int direction;
13       private final double startX;
14       Side(int direction, double startX) {
15           this.direction = direction;
16           this.startX = startX;
17       }
18       public int dir() {
19           return direction;
20       }
21       public double startX() {
22           return startX;
23       }
24       public Side opposite() {
25           return this == LEFT ? RIGHT : LEFT;
26       }
27   }
