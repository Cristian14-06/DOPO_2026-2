3   /**
4    * Time-based match rule with sudden-death golden goal activation on ties.
5    *
6    * @author Juan Espitia
7    * @author Cristian Salamanca
8    */
9   public class TimeRule implements MatchRule{
10      private int timeRemaining;
11      private int tickCount;
12  
13      /**
14       * Creates a default 1-minute time rule.
15       */
16      public TimeRule(){
17          this.timeRemaining = 60;
18          this.tickCount = 0;
19      }
20  
21      /**
22       * Creates a time rule for the given duration in minutes.
23       *
24       * @param minutes Match duration in minutes (must be positive)
25       * @throws DOPOSoccerException if minutes is not positive
26       */
27      public TimeRule(int minutes) throws DOPOSoccerException{
28          if(minutes <= 0) throw new DOPOSoccerException(DOPOSoccerException.INVALID_MATCH_CONFIG);
29          this.timeRemaining = minutes* 60;
30          this.tickCount = 0;
31      }
32  
33      /**
34       * Advances the match countdown clock and triggers sudden death if time expires on a tie.
35       *
36       * @param match Active match instance
37       */
38      @Override
39      public void onTick(Match match) {
40          if (match.getStatus() == GameStatus.PLAYING) {
41              if (timeRemaining > 0) {
42                  tickCount++;
43                  if (tickCount >= Physics.FPS) {
44                      timeRemaining--;
45                      tickCount = 0;
46                  }
47              } else if (match.getScoreBoard().isTie()) {
48                  match.setState(GameStatus.SUDDEN_DEATH);
49              }
50          }
51      }
52  
53      /**
54       * Checks whether regulation time has expired with a non-tied score.
55       *
56       * @param match Active match instance
57       * @return True if time is up and the score is not tied
58       */
59      @Override
60      public boolean isOver(Match match) {
61          return timeRemaining <= 0 && !match.getScoreBoard().isTie();
62      }
63  
64      /**
65       * Returns the remaining match duration in seconds.
66       *
67       * @param match Active match instance
68       * @return Remaining seconds
69       */
70      @Override
71      public int remainingSeconds(Match match) {
72          return timeRemaining;
73      }
74  }
