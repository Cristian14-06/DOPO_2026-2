14  /**
15   * Renders the soccer ball and its power-shot effects in scaled field coordinates.
16   *
17   * @author Juan Espitia
18   * @author Cristian Salamanca
19   */
20  public class BallRenderer {
21      private final ImageAssets assets;
22      private double spinAngle;
23  
24      /**
25       * Constructs a ball renderer using shared image assets.
26       *
27       * @param assets Loaded image assets
28       */
29      public BallRenderer(ImageAssets assets) {
30          this.assets = assets;
31      }
32  
33      /**
34       * Draws the soccer ball and its power-shot glow when active.
35       *
36       * @param g2d Scaled field graphics context
37       * @param match Read-only match view
38       */
39      public void draw(Graphics2D g2d, MatchView match) {
40          if (match == null) return;
41          BallView ball = match.getBall();
42          if (ball == null) return;
43  
44          for(Circle fake : match.getFakeBalls()){
45              drawBallSprite(g2d, (int) fake.getX(), (int) fake.getY(), (int) fake.getRadius(), false);
46          }
47  
48          
49  
50          GameStatus status = match.getStatus();
51          if ((status == GameStatus.PLAYING || status == GameStatus.SUDDEN_DEATH) && ball.getRadius() > 0) {
52              spinAngle += ball.getHorizontalSpeed() / ball.getRadius();
53          }
54  
55          int bx = (int) ball.getX();
56          int by = (int) ball.getY();
57          int r = (int) ball.getRadius();
58  
59          boolean isPower = ball.hasPowerShot();
60  
61          if (isPower) {
62              g2d.setColor(new Color(255, 200, 0, 120));
63              g2d.fillOval(bx - r - 6, by - r - 6, (r + 6) * 2, (r + 6) * 2);
64          }
65  
66          drawChargingRing(g2d, ball, bx, by, r);
67  
68          AffineTransform prevTransform = g2d.getTransform();
69          g2d.rotate(spinAngle, bx, by);
70          drawBallSprite(g2d, bx, by, r, isPower);
71          g2d.setTransform(prevTransform);
72  
73      }
74  
75      /*
76       * Draws a shrinking energy ring around the ball while a power shot is charging.
77       */
78      private void drawChargingRing(Graphics2D g2d, BallView ball, int bx, int by, int r) {
79          if (!ball.isPowerShotCharging()) return;
80          double progress = Math.max(0.0, Math.min(1.0, ball.getPowerShotChargeProgress()));
81          int ringRadius = (int) Math.round(r + 18.0 * (1.0 - progress) + 2.0);
82          Color pc = Theme.powerColor(ball.getPowerShotCountry());
83          g2d.setColor(new Color(pc.getRed(), pc.getGreen(), pc.getBlue(), 200));
84          g2d.setStroke(new BasicStroke(2.2f));
85          g2d.drawOval(bx - ringRadius, by - ringRadius, ringRadius * 2, ringRadius * 2);
86      }
87  
88      /*
89       * Draws the ball sprite (or geometric fallback) centered at the given coordinates.
90       */
91      private void drawBallSprite(Graphics2D g2d, int bx, int by, int r, boolean isPower) {
92          BufferedImage ballImg = (assets != null) ? assets.getBall() : null;
93          if (ballImg != null) {
94              g2d.drawImage(ballImg, bx - r, by - r, r * 2, r * 2, null);
95          } else {
96              g2d.setColor(isPower ? new Color(255, 240, 200) : Color.WHITE);
97              g2d.fillOval(bx - r, by - r, r * 2, r * 2);
98              g2d.setColor(isPower ? new Color(255, 120, 0) : Color.BLACK);
99              g2d.setStroke(new BasicStroke(isPower ? 1.8f : 1.2f));
100             g2d.drawOval(bx - r, by - r, r * 2, r * 2);
101             g2d.drawLine(bx - r + 2, by, bx + r - 2, by);
102             g2d.drawLine(bx, by - r + 2, bx, by + r - 2);
103         }
104     }
