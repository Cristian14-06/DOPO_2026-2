14  /**
15   * Renders a player's limbs, country colors, active power aura, and status effect indicator.
16   *
17   * @author Juan Espitia
18   * @author Cristian Salamanca
19   */
20  public class PlayerRenderer {
21      private static final Font EFFECT_FONT = new Font("Arial", Font.BOLD, 8);
22      private int animFrame;
23  
24      /**
25       * Draws the given player in scaled field coordinates.
26       *
27       * @param g2d Scaled field graphics context
28       * @param p Read-only player view
29       */
30      public void draw(Graphics2D g2d, PlayerView p) {
31          if (p == null || p.getHeadShape() == null) return;
32          animFrame++;
33  
34          Circle hc = p.getHeadShape();
35          int hx = (int) hc.getX();
36          int hy = (int) hc.getY();
37          int hr = (int) hc.getRadius();
38  
39          Color baseColor = Theme.countryColor(p.getCountry());
40  
41          drawAura(g2d, p, hx, hy, hr);
42  
43          Rectangle br = p.getBodyShape();
44          if (br != null) {
45              g2d.setColor(baseColor.darker());
46              g2d.fillRoundRect((int) br.getX(), (int) br.getY(), (int) br.getWidth(), (int) br.getHeight(), 5, 5);
47              g2d.setColor(Color.BLACK);
48              g2d.setStroke(new BasicStroke(1f));
49              g2d.drawRoundRect((int) br.getX(), (int) br.getY(), (int) br.getWidth(), (int) br.getHeight(), 5, 5);
50          }
51  
52          g2d.setColor(baseColor);
53          g2d.fillOval(hx - hr, hy - hr, hr * 2, hr * 2);
54          g2d.setColor(Color.BLACK);
55          g2d.setStroke(new BasicStroke(1.2f));
56          g2d.drawOval(hx - hr, hy - hr, hr * 2, hr * 2);
57  
58          g2d.setColor(Color.WHITE);
59          int eyeX = (p.getSide() == Side.LEFT) ? hx + hr / 3 : hx - hr / 3;
60          g2d.fillOval(eyeX - 2, hy - 2, 4, 5);
61          g2d.setColor(Color.BLACK);
62          g2d.fillOval(eyeX - (p.getSide() == Side.LEFT ? 1 : 2), hy - 1, 2, 3);
63  
64          Circle fc = p.getFootShape();
65          if (fc != null) {
66              int fx = (int) fc.getX();
67              int fy = (int) fc.getY();
68              int fr = (int) fc.getRadius();
69              g2d.setColor(Color.DARK_GRAY);
70              g2d.fillOval(fx - fr, fy - fr, fr * 2, fr * 2);
71              g2d.setColor(Color.BLACK);
72              g2d.drawOval(fx - fr, fy - fr, fr * 2, fr * 2);
73          }
74  
75          drawStatusEffect(g2d, p, hx, hy, hr);
76      }
77  
78      /*
79       * Draws a pulsing country-colored aura around the player while a power is active.
80       */
81      private void drawAura(Graphics2D g2d, PlayerView p, int hx, int hy, int hr) {
82          if (!p.hasAura()) return;
83          double pulse = 0.5 + 0.5 * Math.sin(animFrame * 0.25);
84          int pad = (int) Math.round(7 + pulse * 5);
85          int fillAlpha = (int) Math.round(55 + pulse * 55);
86          Color pc = Theme.powerColor(p.getCountry());
87  
88          g2d.setColor(new Color(pc.getRed(), pc.getGreen(), pc.getBlue(), fillAlpha));
89          g2d.fillOval(hx - hr - pad, hy - hr - pad, (hr + pad) * 2, (hr + pad) * 2);
90          g2d.setColor(new Color(pc.getRed(), pc.getGreen(), pc.getBlue(), 190));
91          g2d.setStroke(new BasicStroke(2f));
92          g2d.drawOval(hx - hr - pad + 2, hy - hr - pad + 2, (hr + pad - 2) * 2, (hr + pad - 2) * 2);
93      }
94  
95      /*
96       * Draws the pulsing head tint, short status symbol, and remaining duration bar for active effects.
97       */
98      private void drawStatusEffect(Graphics2D g2d, PlayerView p, int hx, int hy, int hr) {
99          PlayerEffect effect = p.getActiveEffect();
100         String effectName = p.getActiveEffectName();
101         boolean hasEffect = p.isStunned()
102                 || (effect != null && effect != PlayerEffect.NONE)
103                 || (effectName != null && !effectName.isEmpty());
104         if (!hasEffect) return;
105 
106         Color ec = Theme.effectColor(effect);
107 
108         double pulse = 0.5 + 0.5 * Math.sin(animFrame * 0.2);
109         int alpha = (int) (90 + 60 * pulse);
110         g2d.setColor(new Color(ec.getRed(), ec.getGreen(), ec.getBlue(), alpha));
111         g2d.fillOval(hx - hr, hy - hr, hr * 2, hr * 2);
112 
113         String symbol = getEffectSymbol(effect, p.getSide());
114         g2d.setFont(EFFECT_FONT);
115         int textX = hx - g2d.getFontMetrics().stringWidth(symbol) / 2;
116         int textY = hy - hr - 10;
117         g2d.setColor(new Color(0, 0, 0, 170));
118         g2d.drawString(symbol, textX + 1, textY + 1);
119         g2d.setColor(ec);
120         g2d.drawString(symbol, textX, textY);
121 
122         double progress = Math.max(0.0, Math.min(1.0, p.getEffectProgress()));
123         if (progress > 0.0) {
124             int barW = 28;
125             int barH = 3;
126             int barX = hx - barW / 2;
127             int barY = hy - hr - 6;
128             g2d.setColor(new Color(0, 0, 0, 150));
129             g2d.fillRect(barX, barY, barW, barH);
130             g2d.setColor(ec);
131             g2d.fillRect(barX, barY, (int) Math.round(barW * progress), barH);
132         }
133     }
134 
135     /*
136      * Resolves the compact indicator symbol for the active status effect or stun state.
137      */
138     private String getEffectSymbol(PlayerEffect effect, Side side) {
139         if (effect == null || effect == PlayerEffect.NONE) {
140             return "***";
141         }
142         switch (effect) {
143             case PUSHED:
144                 return (side == Side.LEFT) ? "<<" : ">>";
145             case ELECTROCUTED:
146                 return "!!";
147             case KNOCKED_OUT:
148                 return "Z z";
149             case CONFUSED:
150                 return "?";
151             default:
152                 return "***";
153         }
154     }
