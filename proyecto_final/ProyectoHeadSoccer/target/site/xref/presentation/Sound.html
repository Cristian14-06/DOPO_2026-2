3   /**
4    * Enumerates the sound clips of the game: effects played in response to match events,
5    * and the background soundtrack, which loops while the game is open.
6    *
7    * @author Juan Espitia
8    * @author Cristian Salamanca
9    */
10  public enum Sound {
11      GOAL("goal.wav"),
12      STUNNED("stunned.wav"),
13      FINISH("finish.wav"),
14      KICK("kick.wav"),
15      POWER_AURA("power_aura.wav"),
16      POWER_SHOT("power_shot.wav"),
17      POWER_HIT("power_hit.wav"),
18      SOUNDTRACK("soundtrack.wav");
19  
20      private final String name;
21  
22      /*
23       * Constructs a sound enum constant mapped to its WAV resource filename.
24       */
25      Sound(String name) {
26          this.name = name;
27      }
28  
29      /**
30       * Returns the WAV resource filename for this sound effect.
31       *
32       * @return Audio filename
33       */
34      public String getFileName() {
35          return this.name;
36      }
37  }
