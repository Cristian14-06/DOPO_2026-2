500     /**
501      * Verifies that MatchListener events onPowerActivated, onBallKicked, onPowerShot, and onPowerHit
502      * fire once each at the expected moments during a match.
503      *
504      * @throws DOPOSoccerException if match configuration is invalid
505      */
506     @Test
507     public void testMatchEventsFireOnceAtExpectedMoments() throws DOPOSoccerException {
508         DOPOSoccer game = new DOPOSoccer();
509         MatchConfig config = MatchConfig.builder()
510                 .modality(Modality.PVP)
511                 .leftCountry(Country.SOUTH_KOREA)
512                 .rightCountry(Country.CAMEROON)
513                 .matchType(MatchType.TIME)
514                 .targetValue(2)
515                 .build();
516 
517         AtomicInteger powerActivatedCount = new AtomicInteger(0);
518         AtomicInteger ballKickedCount = new AtomicInteger(0);
519         AtomicInteger powerShotCount = new AtomicInteger(0);
520         AtomicInteger powerHitCount = new AtomicInteger(0);
521         AtomicReference<Country> shotCountry = new AtomicReference<>();
522         AtomicReference<Side> hitVictim = new AtomicReference<>();
523         AtomicReference<Double> hitDamage = new AtomicReference<>(0.0);
524 
525         game.addListener(new MatchListener() {
526             @Override
527             public void onPowerActivated(Side side) {
528                 if (side == Side.LEFT) powerActivatedCount.incrementAndGet();
529             }
530 
531             @Override
532             public void onBallKicked(Side kicker) {
533                 if (kicker == Side.LEFT) ballKickedCount.incrementAndGet();
534             }
535 
536             @Override
537             public void onPowerShot(Side shooter, Country country) {
538                 if (shooter == Side.LEFT) {
539                     powerShotCount.incrementAndGet();
540                     shotCountry.set(country);
541                 }
542             }
543 
544             @Override
545             public void onPowerHit(Side victim, double damage) {
546                 powerHitCount.incrementAndGet();
547                 hitVictim.set(victim);
548                 hitDamage.set(damage);
549             }
550         });
551 
552         game.newMatch(config);
553 
554         for (int i = 0; i < 605; i++) {
555             game.tick();
556         }
557         assertEquals(0, powerActivatedCount.get());
558 
559         game.setIntention(Side.LEFT, EnumSet.of(PlayerAction.POWER));
560         game.tick();
561         game.tick();
562         assertEquals(1, powerActivatedCount.get());
563 
564         for (int i = 0; i < 120 && powerShotCount.get() == 0; i++) {
565             double px = game.getMatch().getPlayer(Side.LEFT).getFootShape().getX();
566             double bx = game.getMatch().getBall().getX();
567             if (Math.abs(bx - px) < 28.0) {
568                 game.setIntention(Side.LEFT, EnumSet.of(PlayerAction.RIGHT, PlayerAction.KICK));
569             } else {
570                 game.setIntention(Side.LEFT, EnumSet.of(PlayerAction.RIGHT));
571             }
572             game.tick();
573         }
574 
575         assertEquals(1, powerShotCount.get());
576         assertEquals(Country.SOUTH_KOREA, shotCountry.get());
577         assertTrue(ballKickedCount.get() >= 1);
578 
579         game.setIntention(Side.LEFT, EnumSet.noneOf(PlayerAction.class));
580         for (int i = 0; i < Power.TICKS_PER_POWER + 60 && powerHitCount.get() == 0; i++) {
581             game.tick();
582         }
583 
584         assertEquals(1, powerHitCount.get());
585         assertEquals(Side.RIGHT, hitVictim.get());
586         assertTrue(hitDamage.get() > 0.0);
587         assertFalse(game.getMatch().getBall().hasPowerShot());
588     }
