32      /**
33       * Verifies that MatchSetup.toConfig builds valid PvP and PvM configurations,
34       * applies TIME and GOALS target values, and rejects duplicate player names.
35       *
36       * @throws DOPOSoccerException if configuration validation fails unexpectedly
37       */
38      @Test
39      public void testMatchSetupToConfigPvPAndPvMAndDuplicateValidation() throws DOPOSoccerException {
40          MatchSetup setup = new MatchSetup();
41          setup.setLeftPlayerName("  Juan  ");
42          setup.setRightPlayerName("  Cristian ");
43          setup.setLeftPlayerName("   ");
44          assertEquals("Juan", setup.getLeftPlayerName());
45          assertEquals("Cristian", setup.getRightPlayerName());
46  
47          MatchConfig pvpConfig = setup.toConfig();
48          assertEquals(Modality.PVP, pvpConfig.getModality());
49          assertEquals("Juan", pvpConfig.getLeftPlayerName());
50          assertEquals("Cristian", pvpConfig.getRightPlayerName());
51          assertEquals(MatchType.TIME, pvpConfig.getMatchType());
52          assertEquals(2, pvpConfig.getTargetValue());
53  
54          setup.setModality(Modality.PVC);
55          setup.selectOrCycleGoalsMatch();
56          MatchConfig pvmConfig = setup.toConfig();
57          assertEquals(Modality.PVC, pvmConfig.getModality());
58          assertEquals("Juan", pvmConfig.getLeftPlayerName());
59          assertEquals("COM " + setup.getMachineProfile().getDisplayName(), pvmConfig.getRightPlayerName());
60          assertEquals(MatchType.GOALS, pvmConfig.getMatchType());
61          assertEquals(3, pvmConfig.getTargetValue());
62  
63          setup.setModality(Modality.PVP);
64          setup.setRightPlayerName("Juan");
65          try {
66              setup.toConfig();
67              fail("Expected DOPOSoccerException when both player names are equal");
68          } catch (DOPOSoccerException ex) {
69              assertEquals(DOPOSoccerException.INVALID_MATCH_CONFIG, ex.getMessage());
70          }
71      }

92          MachineProfile initialProfile = setup.getMachineProfile();

98          setup.selectOrCycleTimeMatch();

100         setup.selectOrCycleTimeMatch();

122     /**
123      * Verifies that Theme color lookups return valid colors for every enum value and null,
124      * and that Swing styling helpers configure button properties cleanly.
125      */
126     @Test
127     public void testThemeColorsAndButtonHelpers() {

131         }
132         assertNotNull(Theme.countryColor(null));
133         assertNotNull(Theme.powerColor(null));
134 
135         for (PlayerEffect e : PlayerEffect.values()) {
136             assertNotNull(Theme.effectColor(e));
137         }
138         assertEquals(Color.WHITE, Theme.effectColor(null));
139 
140         JButton btn = Theme.createStyledButton("PLAY", Theme.ACTION_BLUE, Color.WHITE, 16);
141         assertEquals("PLAY", btn.getText());
142         assertEquals(Theme.ACTION_BLUE, btn.getBackground());
143         assertEquals(Color.WHITE, btn.getForeground());
144 
145         Theme.styleToggle(btn, true);
146         assertEquals(Theme.ACCENT_GOLD, btn.getBackground());
147         assertEquals(Color.BLACK, btn.getForeground());
148 
149         Theme.styleToggle(btn, false);
150         assertEquals(Theme.TOGGLE_UNSELECTED, btn.getBackground());
151         assertEquals(Color.WHITE, btn.getForeground());
152     }
